<script id="grafikaJsSource" type="text/plain">
/**
 * ============================================================================
 * GTA SUS 3D: INFINITY EDITION — 64-BIT SILNIK GRAFICZNY (grafika.js)
 * ============================================================================
 * Moduł graficzny klasy 64-bit (precyzja Float64 + indeksowanie sektorów
 * BigInt64, proceduralne tekstury Canvas PBR, oświetlenie dzienne / złota
 * godzina / nocne okna biurowców / powierzchnia Księżyca NASA, wielowarstwowe
 * chmury 3D, tonowanie ACES Filmic, cienie PCFSoftShadowMap, pełne
 * zwalnianie zasobów GPU oraz hooki dla index.html i GTA_SUS_OPTIMIZER).
 * ============================================================================
 */
(function (global) {
  'use strict';

  const SECTOR_SIZE_METERS = 60.0;

  // ==========================================================================
  // 1. NARZĘDZIA MATEMATYCZNE I SEKTOROWE 64-BIT (Float64 + BigInt64)
  // ==========================================================================
  const Math64 = {
    clamp64(value, min, max, fallback = min) {
      const n = Number(value);
      if (!Number.isFinite(n)) return Number(fallback);
      if (n < min) return Number(min);
      if (n > max) return Number(max);
      return n;
    },

    lerp64(a, b, t) {
      const k = Math64.clamp64(t, 0, 1, 0);
      return Number(a) + (Number(b) - Number(a)) * k;
    },

    hashSeed64(seedInput) {
      const str = String(seedInput || 'DUBAI-NYC-64BIT-2026');
      let hash = 0xcbf29ce484222325n;
      const prime = 0x100000001b3n;
      const mask64 = 0xffffffffffffffffn;
      for (let i = 0; i < str.length; i++) {
        hash ^= BigInt(str.charCodeAt(i) & 0xff);
        hash = (hash * prime) & mask64;
      }
      return hash === 0n ? 0x64b1700000000001n : hash;
    },

    createRng64(seedInput) {
      let state = Math64.hashSeed64(seedInput);
      const mask64 = 0xffffffffffffffffn;
      return function next64() {
        state = (state + 0x9e3779b97f4a7c15n) & mask64;
        let z = state;
        z = ((z ^ (z >> 30n)) * 0xbf58476d1ce4e5b9n) & mask64;
        z = ((z ^ (z >> 27n)) * 0x94d049bb133111ebn) & mask64;
        z = z ^ (z >> 31n);
        return Number(z >> 11n) / 9007199254740992; // pełne 53 bity mantysy Float64
      };
    },

    worldToSector64(x, z, sectorSize = SECTOR_SIZE_METERS) {
      const sx = BigInt(Math.floor(Number(x || 0) / sectorSize));
      const sz = BigInt(Math.floor(Number(z || 0) / sectorSize));
      const key64 = ((sx & 0xffffffffn) << 32n) | (sz & 0xffffffffn);
      return { sx, sz, key64 };
    },

    formatCoords64(x, y, z) {
      const fx = Number(x || 0).toFixed(8);
      const fy = Number(y || 0).toFixed(8);
      const fz = Number(z || 0).toFixed(8);
      return `X:${fx} Y:${fy} Z:${fz}`;
    }
  };

  // ==========================================================================
  // 2. PROCEDURALNY GENERATOR TEKSTUR PBR (OKNA, ASFALT, WODA, KSIĘŻYC, PARK)
  // ==========================================================================
  class ProceduralTextureBank64 {
    constructor(THREE) {
      this.THREE = THREE;
      this.cache = new Map();
    }

    _makeCanvas(w, h) {
      if (typeof document === 'undefined') return null;
      const c = document.createElement('canvas');
      c.width = w;
      c.height = h;
      return c;
    }

    _wrapTexture(canvas, repeatX = 1, repeatY = 1, isColor = true) {
      const THREE = this.THREE;
      if (!canvas || !THREE || !THREE.CanvasTexture) return null;
      const tex = new THREE.CanvasTexture(canvas);
      tex.wrapS = THREE.RepeatWrapping;
      tex.wrapT = THREE.RepeatWrapping;
      tex.repeat.set(repeatX, repeatY);
      if (isColor && THREE.sRGBEncoding) {
        tex.encoding = THREE.sRGBEncoding;
      }
      tex.needsUpdate = true;
      return tex;
    }

    getSkyscraperFacadeTextures(mode = 'day', seedVal = 1) {
      const key = `facade_${mode}_${seedVal}`;
      if (this.cache.has(key)) return this.cache.get(key);

      const canvas = this._makeCanvas(256, 512);
      if (!canvas) return { map: null, emissiveMap: null, roughnessMap: null };

      const ctx = canvas.getContext('2d');
      const emCanvas = this._makeCanvas(256, 512);
      const emCtx = emCanvas.getContext('2d');
      const rng = Math64.createRng64(`${seedVal}-${mode}-facade64`);

      // Szkło architektoniczne i stal Dubaj/USA
      ctx.fillStyle = mode === 'night' ? '#091124' : mode === 'golden_hour' ? '#1e293b' : '#16243b';
      ctx.fillRect(0, 0, 256, 512);

      emCtx.fillStyle = '#000000';
      emCtx.fillRect(0, 0, 256, 512);

      const cols = 12;
      const rows = 36;
      const padX = 5;
      const padY = 4;
      const cellW = (256 - padX * (cols + 1)) / cols;
      const cellH = (512 - padY * (rows + 1)) / rows;

      const litProbability =
        mode === 'night' ? 0.72 : mode === 'golden_hour' ? 0.44 : 0.16;

      for (let r = 0; r < rows; r++) {
        const floorWarmth = rng();
        for (let c = 0; c < cols; c++) {
          const x = padX + c * (cellW + padX);
          const y = padY + r * (cellH + padY);
          const isLit = rng() < litProbability;

          if (isLit) {
            const warm = floorWarmth > 0.35;
            ctx.fillStyle = warm ? '#fef08a' : '#bae6fd';
            emCtx.fillStyle = warm ? '#fbbf24' : '#7dd3fc';
            emCtx.fillRect(x, y, cellW, cellH);
          } else {
            ctx.fillStyle =
              mode === 'night'
                ? '#0f172a'
                : mode === 'golden_hour'
                ? '#38bdf8'
                : '#38bdf8';
          }
          ctx.fillRect(x, y, cellW, cellH);
        }
      }

      // Pionowe żebra architektoniczne
      ctx.fillStyle = 'rgba(148, 163, 184, 0.22)';
      for (let c = 0; c <= cols; c++) {
        ctx.fillRect(c * (cellW + padX) + 1, 0, 2.5, 512);
      }

      const map = this._wrapTexture(canvas, 1, 2, true);
      const emissiveMap = this._wrapTexture(emCanvas, 1, 2, true);
      const bundle = { map, emissiveMap };
      this.cache.set(key, bundle);
      return bundle;
    }

    getAsphaltTexture(seedVal = 1) {
      const key = `asphalt_${seedVal}`;
      if (this.cache.has(key)) return this.cache.get(key);

      const canvas = this._makeCanvas(256, 256);
      if (!canvas) return null;
      const ctx = canvas.getContext('2d');
      const rng = Math64.createRng64(`asphalt64_${seedVal}`);

      ctx.fillStyle = '#111827';
      ctx.fillRect(0, 0, 256, 256);

      for (let i = 0; i < 4200; i++) {
        const x = Math.floor(rng() * 256);
        const y = Math.floor(rng() * 256);
        const v = 22 + Math.floor(rng() * 26);
        ctx.fillStyle = `rgb(${v},${v + 2},${v + 6})`;
        ctx.fillRect(x, y, 2, 2);
      }

      const tex = this._wrapTexture(canvas, 28, 28, true);
      this.cache.set(key, tex);
      return tex;
    }

    getWaterTexture(seedVal = 1) {
      const key = `water_${seedVal}`;
      if (this.cache.has(key)) return this.cache.get(key);

      const canvas = this._makeCanvas(256, 256);
      if (!canvas) return null;
      const ctx = canvas.getContext('2d');
      const rng = Math64.createRng64(`water64_${seedVal}`);

      ctx.fillStyle = '#0369a1';
      ctx.fillRect(0, 0, 256, 256);

      ctx.strokeStyle = 'rgba(186, 230, 253, 0.24)';
      ctx.lineWidth = 2;
      for (let i = 0; i < 42; i++) {
        const y = rng() * 256;
        const x = rng() * 256;
        const len = 18 + rng() * 34;
        ctx.beginPath();
        ctx.moveTo(x, y);
        ctx.quadraticCurveTo(x + len * 0.5, y - 4, x + len, y);
        ctx.stroke();
      }

      const tex = this._wrapTexture(canvas, 8, 8, true);
      this.cache.set(key, tex);
      return tex;
    }

    getMoonSurfaceTexture(seedVal = 1) {
      const key = `moon_${seedVal}`;
      if (this.cache.has(key)) return this.cache.get(key);

      const canvas = this._makeCanvas(512, 512);
      if (!canvas) return null;
      const ctx = canvas.getContext('2d');
      const rng = Math64.createRng64(`moon64_${seedVal}`);

      ctx.fillStyle = '#64748b';
      ctx.fillRect(0, 0, 512, 512);

      for (let i = 0; i < 2500; i++) {
        const x = rng() * 512;
        const y = rng() * 512;
        const shade = 75 + Math.floor(rng() * 65);
        ctx.fillStyle = `rgb(${shade},${shade + 4},${shade + 10})`;
        ctx.fillRect(x, y, 2, 2);
      }

      for (let c = 0; c < 48; c++) {
        const cx = rng() * 512;
        const cy = rng() * 512;
        const cr = 5 + rng() * 24;
        ctx.strokeStyle = 'rgba(30, 41, 59, 0.55)';
        ctx.lineWidth = 2.5;
        ctx.beginPath();
        ctx.arc(cx, cy, cr, 0, Math.PI * 2);
        ctx.stroke();
      }

      const tex = this._wrapTexture(canvas, 18, 18, true);
      this.cache.set(key, tex);
      return tex;
    }

    disposeAll() {
      this.cache.forEach((entry) => {
        if (!entry) return;
        if (entry.map && entry.map.dispose) entry.map.dispose();
        if (entry.emissiveMap && entry.emissiveMap.dispose) entry.emissiveMap.dispose();
        if (entry.dispose) entry.dispose();
      });
      this.cache.clear();
    }
  }

  // ==========================================================================
  // 3. GŁÓWNA KLASA SILNIKA GRAFICZNEGO 64-BIT: GtaSusGraphics64
  // ==========================================================================
  class GtaSusGraphics64 {
    constructor(options = {}) {
      this.THREE = options.THREE || global.THREE;
      this.renderer = options.renderer || null;
      this.scene = options.scene || null;
      this.camera = options.camera || null;
      this.dirLight = options.dirLight || null;
      this.hemiLight = options.hemiLight || null;

      this.config = {
        preset: options.preset || 'day',
        quality: options.quality || 'high',
        exposure: Math64.clamp64(options.exposure ?? 1.1, 0.65, 1.65, 1.1),
        cloudDensity: Math64.clamp64(options.cloudDensity ?? 80, 0, 100, 80),
        fogDensity: Math64.clamp64(options.fogDensity ?? 0.0011, 0.0002, 0.0055, 0.0011),
        bloomIntensity: Math64.clamp64(options.bloomIntensity ?? 0.26, 0, 0.85, 0.26),
        seed: String(options.seed || 'DUBAI-NYC-64BIT-2026')
      };

      this.textures = this.THREE ? new ProceduralTextureBank64(this.THREE) : null;
      this.managedGroup = null;
      this.cloudLayers = [];
      this.waterMeshes = [];
      this.moonGroup = null;
      this.sunDisc = null;
      this.elapsedSec = 0;

      this._onPostFxChange = typeof options.onPostFxChange === 'function' ? options.onPostFxChange : null;
    }

    attach(context = {}) {
      if (context.renderer) this.renderer = context.renderer;
      if (context.scene) this.scene = context.scene;
      if (context.camera) this.camera = context.camera;
      if (context.dirLight) this.dirLight = context.dirLight;
      if (context.hemiLight) this.hemiLight = context.hemiLight;
      if (context.state && context.state.seed) {
        this.config.seed = String(context.state.seed);
      }

      if (!this.textures && (context.THREE || global.THREE)) {
        this.THREE = context.THREE || global.THREE;
        this.textures = new ProceduralTextureBank64(this.THREE);
      }

      this.configureRenderer64();
      this.applyEnvironmentPreset(this.config.preset);
      return this;
    }

    configureRenderer64() {
      const THREE = this.THREE;
      const r = this.renderer;
      if (!r || !THREE) return;

      const qualityMap = {
        low: { shadowSize: 512, maxDpr: 1.0, shadows: false },
        medium: { shadowSize: 1024, maxDpr: 1.35, shadows: true },
        high: { shadowSize: 2048, maxDpr: 1.75, shadows: true },
        ultra: { shadowSize: 4096, maxDpr: 2.0, shadows: true }
      };
      const q = qualityMap[this.config.quality] || qualityMap.high;

      r.toneMapping = THREE.ACESFilmicToneMapping;
      r.toneMappingExposure = Math64.clamp64(this.config.exposure, 0.65, 1.65, 1.1);
      r.shadowMap.enabled = q.shadows;
      r.shadowMap.type = THREE.PCFSoftShadowMap;

      if (typeof window !== 'undefined' && typeof r.setPixelRatio === 'function') {
        const opt = global.GTA_SUS_OPTIMIZER || global.optymalizer;
        const drsFactor = opt && typeof opt.scale === 'number' ? Math64.clamp64(opt.scale, 0.5, 1.0, 1.0) : 1.0;
        const targetDpr = Math.min(window.devicePixelRatio || 1, q.maxDpr) * drsFactor;
        r.setPixelRatio(Math.max(0.5, targetDpr));
      }

      if (this.dirLight && this.dirLight.shadow) {
        this.dirLight.castShadow = q.shadows;
        const mapSize = this.dirLight.shadow.mapSize;
        if (mapSize && mapSize.width !== q.shadowSize) {
          mapSize.set(q.shadowSize, q.shadowSize);
          if (this.dirLight.shadow.map) {
            this.dirLight.shadow.map.dispose();
            this.dirLight.shadow.map = null;
          }
        }
      }
    }

    applyEnvironmentPreset(presetName) {
      const THREE = this.THREE;
      if (!THREE || !this.scene) return;

      const validPresets = ['day', 'golden_hour', 'night', 'moon'];
      const preset = validPresets.includes(presetName) ? presetName : 'day';
      this.config.preset = preset;

      const palettes = {
        day: {
          bg: 0x5aa9e6,
          fog: 0x8ecae6,
          fogDensity: this.config.fogDensity,
          hemiSky: 0xe0f2fe,

    updateCloudVisibility(tintHex = null) {
      const THREE = this.THREE;
      if (!THREE || this.cloudLayers.length === 0) return;

      const isMoon = this.config.preset === 'moon';
      const activeRatio = isMoon ? 0 : Math64.clamp64(this.config.cloudDensity / 100, 0, 1, 0.8);
      const visibleCount = Math.round(this.cloudLayers.length * activeRatio);

      const defaultColor =
        this.config.preset === 'night'
          ? 0x1e293b
          : this.config.preset === 'golden_hour'
          ? 0xffedd5
          : 0xffffff;
      const targetHex = tintHex !== null ? tintHex : defaultColor;

      this.cloudLayers.forEach((cloud, idx) => {
        if (!cloud || !cloud.mesh) return;
        cloud.mesh.visible = idx < visibleCount;
        cloud.mesh.traverse((child) => {
          if (child.isMesh && child.material && child.material.color) {
            child.material.color.setHex(targetHex);
          }
        });
      });
    }

      const cloudMat = new THREE.MeshStandardMaterial({ color: 0xffffff, roughness: 0.92, metalness: 0.0 });

            p.scale.set(sc * 1.35, sc * 0.72, sc * 1.15);
            grp.add(p);
          }
        );
        grp.position.set(cx, cy, cz);
        this.managedGroup.add(grp);
        this.cloudLayers.push({
          mesh: grp,
          baseY: cy,
          speed: 2.2 + rng() * 2.6,
          phase: rng() * Math.PI * 2
        });

      this.moonGroup.add(moonFloor);
      this.managedGroup.add(this.moonGroup);

      this.applyEnvironmentPreset(this.config.preset);
      return this.managedGroup;
    }

    update(dtSec = 0.016, context = {}) {
      const dt = Math64.clamp64(dtSec, 0.001, 0.1, 0.016);
      this.elapsedSec += dt;

      if (context.renderer) this.renderer = context.renderer;
      if (context.scene) this.scene = context.scene;
      if (context.camera) this.camera = context.camera;
      if (context.dirLight) this.dirLight = context.dirLight;
      if (context.hemiLight) this.hemiLight = context.hemiLight;

      this.configureRenderer64();

      // Dryf wielowarstwowych chmur 3D
      for (let i = 0; i < this.cloudLayers.length; i++) {
        const c = this.cloudLayers[i];
        if (!c || !c.mesh || !c.mesh.visible) continue;
        c.mesh.position.x += c.speed * dt;
        c.mesh.position.y = c.baseY + Math.sin(this.elapsedSec * 0.85 + c.phase) * 0.9;
        if (c.mesh.position.x > 320) {
          c.mesh.position.x = -320;
        }
      }

      // Animacja fal zatoki
      if (this.waterMeshes.length > 0 && this.textures) {
        const wTex = this.textures.getWaterTexture(Number(Math64.hashSeed64(this.config.seed) % 100000n));
        if (wTex) {
          wTex.offset.x = (this.elapsedSec * 0.03) % 1;
          wTex.offset.y = (this.elapsedSec * 0.02) % 1;
        }
      }
    }

    setSettings(patch = {}) {
      if (patch.preset !== undefined) this.config.preset = String(patch.preset);
      if (patch.quality !== undefined) this.config.quality = String(patch.quality);
      if (patch.exposure !== undefined) {
        this.config.exposure = Math64.clamp64(patch.exposure, 0.65, 1.65, this.config.exposure);
      }
      if (patch.cloudDensity !== undefined) {
        this.config.cloudDensity = Math64.clamp64(patch.cloudDensity, 0, 100, this.config.cloudDensity);
      }
      if (patch.fogDensity !== undefined) {
        this.config.fogDensity = Math64.clamp64(patch.fogDensity, 0.0002, 0.0055, this.config.fogDensity);
      }
      if (patch.bloomIntensity !== undefined) {
        this.config.bloomIntensity = Math64.clamp64(patch.bloomIntensity, 0, 0.85, this.config.bloomIntensity);
      }
      if (patch.seed !== undefined) {
        this.config.seed = String(patch.seed);
      }
      this.applyEnvironmentPreset(this.config.preset);
    }

    _emitPostFx() {
      if (this._onPostFxChange) {
        this._onPostFxChange({
          bloomIntensity: this.config.bloomIntensity,
          preset: this.config.preset,
          exposure: this.config.exposure
        });
      }
    }

    clearManagedScene() {
      if (this.managedGroup) {
        if (this.managedGroup.parent) {
          this.managedGroup.parent.remove(this.managedGroup);
        }
        this.managedGroup.traverse((child) => {
          if (child.geometry && typeof child.geometry.dispose === 'function') {
            child.geometry.dispose();
          }
          if (child.material) {
            const mats = Array.isArray(child.material) ? child.material : [child.material];
            mats.forEach((m) => {
              if (m && typeof m.dispose === 'function') m.dispose();
            });
          }
        });
        this.managedGroup = null;
      }
      this.cloudLayers = [];
      this.waterMeshes = [];
      this.moonGroup = null;
      this.sunDisc = null;
    }

    dispose() {
      this.clearManagedScene();
      if (this.textures) {
        this.textures.disposeAll();
      }
    }
  }

  const instance = new GtaSusGraphics64();
  global.GtaSusGraphics64 = GtaSusGraphics64;
  global.Math64 = Math64;
  global.GTA_SUS_GRAFIKA = instance;
  global.grafika = instance;

  if (global.GTA_SUS_ENGINE && typeof global.GTA_SUS_ENGINE.registerGrafikaModule === 'function') {
    global.GTA_SUS_ENGINE.registerGrafikaModule(instance);
  }
})(typeof window !== 'undefined' ? window : globalThis);
</script>

      dragStartY = e.clientY;
    });
    window.addEventListener('mouseup', () => { isDragging = false; });
    window.addEventListener('mousemove', (e) => {
      if (!isDragging) return;
      const dx = e.clientX - dragStartX;
      const dy = e.clientY - dragStartY;
      dragStartX = e.clientX;
      dragStartY = e.clientY;
      orbitAngle -= dx * 0.006;
      orbitHeight = Math.max(12, Math.min(130, orbitHeight + dy * 0.35));
    });
    canvas.addEventListener('wheel', (e) => {
      orbitRadius = Math.max(38, Math.min(240, orbitRadius + e.deltaY * 0.06));
    }, { passive: true });

    bindControls();
    requestAnimationFrame(renderLoop);
  }

  function bindControls() {

      $('graphicsControlPanel').classList.toggle('hidden');

      gfx.setSettings({ exposure: val });

      gfx.setSettings({ cloudDensity: val });
    });

    $('sliderFog').addEventListener('input', (e) => {
      const val = Number(e.target.value);
      $('valFog').textContent = val.toFixed(4);

    $('sliderBloom').addEventListener('input', (e) => {

      gfx.setSettings({ seed: nextSeed });
      gfx.buildShowcaseScene(nextSeed);

      gfx.setSettings({

      $('sliderExposure').value = '1.10';

      $('valFog').textContent = '0.0011';

      gfx.buildShowcaseScene('DUBAI-NYC-64BIT-2026');
    });
  }

  let prevMs = performance.now();
  function renderLoop(nowMs) {
    const dt = Math.min(0.05, Math.max(0.001, (nowMs - prevMs) / 1000));
    prevMs = nowMs;

    fpsCounter++;
    fpsAccum += dt;
    if (fpsAccum >= 0.4) {
      currentFps = Math.max(1, Math.round(fpsCounter / fpsAccum));
      fpsCounter = 0;
      fpsAccum = 0;
      if ($('liveFpsBadge')) $('liveFpsBadge').textContent = `${currentFps} FPS`;
    }

    if (!isDragging) {
      orbitAngle += 0.07 * dt;
    }

    const camX = Math.cos(orbitAngle) * orbitRadius;
    const camZ = Math.sin(orbitAngle) * orbitRadius;
    camera.position.set(camX, orbitHeight, camZ);
    camera.lookAt(0, 26, -35);

    const gfx = window.GTA_SUS_GRAFIKA;
    if (gfx) {
      gfx.update(dt, { renderer, scene, camera, dirLight, hemiLight });
    }

    if ($('coord64Readout') && window.Math64) {
      $('coord64Readout').textContent = window.Math64.formatCoords64(camX, orbitHeight, camZ);
      const sec = window.Math64.worldToSector64(camX, camZ);
      const objCount = gfx && gfx.managedGroup ? gfx.managedGroup.children.length : 0;
      $('chunk64Readout').textContent = `Sektor BigInt64: (${sec.sx.toString()}, ${sec.sz.toString()}) • Obiekty: ${objCount}`;
    }

    renderer.render(scene, camera);
    requestAnimationFrame(renderLoop);
  }

  initStage();
})();
