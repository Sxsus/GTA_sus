inline double clamp64(double val, double lo, double hi, double fallback) {
    if (std::isnan(val) || std::isinf(val)) {
        return fallback;
    }
    return std::max(lo, std::min(hi, val));
}

inline std::string toLower(std::string s) {
    for (char& ch : s) {
        ch = static_cast<char>(std::tolower(static_cast<unsigned char>(ch)));
    }
    return s;
}

inline std::string escapeJson(const std::string& input) {
    std::ostringstream oss;
    for (unsigned char c : input) {
        switch (c) {
            case '"':  oss << "\\\""; break;
            case '\\': oss << "\\\\"; break;
            case '\n': oss << "\\n";  break;
            case '\r': oss << "\\r";  break;
            case '\t': oss << "\\t";  break;
            default:
                if (c < 0x20) {
                    oss << "\\u" << std::hex << std::setw(4) << std::setfill('0') << static_cast<int>(c)
                        << std::dec;
                } else {
                    oss << static_cast<char>(c);
                }
                break;
        }
    }
    return oss.str();
}

/**
 * 64-bitowy zarządca pamięci i sektorów otwartego świata (styl RAGE memory streaming).
 */
class MemoryArena64 {
public:
    explicit MemoryArena64(double sectorSize = 1024.0)
        : sectorSize_(clamp64(sectorSize, 128.0, 65536.0, 1024.0)) {}

    void setOrigin(double x, double y, double z) {
        origin_[0] = clamp64(x, -1e9, 1e9, 0.0);
        origin_[1] = clamp64(y, -1e6, 1e6, 0.0);
        origin_[2] = clamp64(z, -1e9, 1e9, 0.0);

        sector_[0] = static_cast<int64_t>(std::floor(origin_[0] / sectorSize_));
        sector_[1] = static_cast<int64_t>(std::floor(origin_[1] / sectorSize_));
        sector_[2] = static_cast<int64_t>(std::floor(origin_[2] / sectorSize_));
    }

    void trackAllocation(std::size_t bytes) {
        activeBytes_ += bytes;
        allocationsCount_ += 1;
        if (activeBytes_ > peakBytes_) {
            peakBytes_ = activeBytes_;
        }
    }

    void releaseBytes(std::size_t bytes) {
        if (bytes >= activeBytes_) {
            activeBytes_ = 0;
        } else {
            activeBytes_ -= bytes;
        }
    }

    void flushStreamPool() {
        activeBytes_ = 0;
        flushesCount_ += 1;
    }

    std::string sectorKey() const {
        return std::to_string(sector_[0]) + ":" +
               std::to_string(sector_[1]) + ":" +
               std::to_string(sector_[2]);
    }

    std::size_t activeBytes() const { return activeBytes_; }
    std::size_t peakBytes() const { return peakBytes_; }
    std::size_t allocationsCount() const { return allocationsCount_; }
    std::size_t flushesCount() const { return flushesCount_; }

private:
    double sectorSize_;
    std::array<double, 3> origin_{0.0, 0.0, 0.0};
    std::array<int64_t, 3> sector_{0, 0, 0};
    std::size_t activeBytes_ = 0;
    std::size_t peakBytes_ = 0;
    std::size_t allocationsCount_ = 0;
    std::size_t flushesCount_ = 0;
};

struct WorldActorState {
    int hp = 100;
    long long cash = 1000;
    int stars = 0;
    double altitude = 0.0;
    double vy = 0.0;
    bool inVehicle = false;
    bool parachuteOpen = false;
    bool hasParachute = true;
    bool inWater = false;
    bool onMoon = false;
    std::string vehicleType = "car";

    // Pola wyjściowe po symulacji / naprawie
    bool blockCarInWater = false;
    double euphoriaReactionForce = 0.0;
    std::vector<std::string> fixes;
};

/**
 * Fizyka lotu na Spacji, blokada wjazdu aut do wody, obrażenia od upadku + spadochron
 * oraz reakcje fizyczne postaci w stylu Euphoria.
 */
class RagePhysicsEuphoria {
public:
    double spacebarClimbRate = 28.0;
    double helicopterLiftAccel = 34.0;
    double parachuteDescentSpeed = 5.0;
    double fallDamageThreshold = 11.5;
    double fallDamageMultiplier = 3.2;
    double moonGravityScale = 0.24;
    double euphoriaImpulseScale = 1.25;

    void step(WorldActorState& s, double dt, bool spaceHeld, bool descendHeld) const {
        const double safeDt = clamp64(dt, 0.001, 0.25, 0.016);

        // 1. Naprawa wartości brzegowych
        if (s.hp < 0 || s.hp > 100) {
            int oldHp = s.hp;
            s.hp = static_cast<int>(clamp64(s.hp, 0.0, 100.0, 100.0));
            s.fixes.push_back("RAGE/Euphoria: skorygowano HP (" + std::to_string(oldHp) + " -> " + std::to_string(s.hp) + ").");
        }
        if (s.cash < 0) {
            long long oldCash = s.cash;
            s.cash = 0;
            s.fixes.push_back("RAGE: naprawiono ujemną gotówkę ($" + std::to_string(oldCash) + " -> $0).");
        }
        if (s.stars < 0 || s.stars > 6) {
            int oldStars = s.stars;
            s.stars = static_cast<int>(clamp64(s.stars, 0.0, 6.0, 0.0));
            s.fixes.push_back("RAGE: skorygowano poziom pościgu (" + std::to_string(oldStars) + " -> " + std::to_string(s.stars) + "★).");
        }

        const std::string vType = toLower(s.vehicleType);
        const bool isBoat = (vType == "boat" || vType == "ship" || vType == "jet_ski");
        const bool isPlane = (vType == "plane" || vType == "airplane");
        const bool isHeli = (vType == "heli" || vType == "helicopter");

        // 2. Blokada wjazdu pojazdów lądowych do głębokiej wody
        s.blockCarInWater = false;
        if (s.inVehicle && s.inWater && !isBoat) {
            s.blockCarInWater = true;
            s.fixes.push_back("RAGE: zablokowano wjazd pojazdu lądowego do wody (użyj łodzi lub mostu).");
        }

        // 3. Lot samolotem i helikopterem na Spacji
        if (s.inVehicle && (isPlane || isHeli)) {
            if (spaceHeld) {
                const double lift = isHeli ? helicopterLiftAccel : spacebarClimbRate;
                s.vy += lift * safeDt;
            } else if (descendHeld) {
                s.vy -= 18.0 * safeDt;
            } else if (isHeli) {
                s.vy *= 0.92;
            }
            s.altitude = std::max(0.0, s.altitude + s.vy * safeDt);
        } else if (!s.inVehicle) {
            // 4. Grawitacja, spadochron i obrażenia od upadku
            const double grav = s.onMoon ? (9.81 * moonGravityScale) : 9.81;
            if (s.altitude > 0.0 || s.vy > 0.0) {
                if (!s.parachuteOpen && s.altitude > 14.0 && s.vy < -14.0 && s.hasParachute && spaceHeld) {
                    s.parachuteOpen = true;
                    s.fixes.push_back("RAGE: otwarto spadochron w locie pod Spacją.");
                }
                if (s.parachuteOpen) {
                    s.vy = std::max(-parachuteDescentSpeed, s.vy - grav * 0.35 * safeDt);
                } else {
                    s.vy -= grav * safeDt;
                }

                s.altitude += s.vy * safeDt;
                if (s.altitude <= 0.0) {
                    const double impactSpeed = std::abs(s.vy);
                    s.altitude = 0.0;
                    s.vy = 0.0;
                    if (!s.parachuteOpen && impactSpeed > fallDamageThreshold) {
                        const int dmg = static_cast<int>(std::round((impactSpeed - fallDamageThreshold) * fallDamageMultiplier));
                        s.hp = std::max(0, s.hp - dmg);
                        s.euphoriaReactionForce = impactSpeed * euphoriaImpulseScale;
                        s.fixes.push_back("Euphoria: obrażenia od upadku (-" + std::to_string(dmg) + " HP).");
                    }
                    s.parachuteOpen = false;
                }
            }
        }

        // 5. Siła reakcji Euphoria przy pościgu lub kolizji
        s.euphoriaReactionForce = std::round((std::abs(s.vy) * 0.4 + s.stars * 12.5) * euphoriaImpulseScale * 100.0) / 100.0;
    }
};

/**
 * Integracja z Groq AI (z nowym kluczem na czele listy), wzmocnienie funkcji
 * oraz eksport profilu do gta_cpp_tuning.js.
 */
class GroqRageDirector {
public:
    explicit GroqRageDirector(std::string seed = "DUBAI-SUS-64")
        : seed_(seed.empty() ? "DUBAI-SUS-64" : seed), keys_(kDefaultGroqKeys) {
        const char* envKey = std::getenv("GROQ_API_KEY");
        if (envKey && *envKey) {
            keys_.insert(keys_.begin(), std::string(envKey));
        }
    }

    std::size_t keyCount() const { return keys_.size(); }
    const std::string& activeKeyPreview() const {
        return keys_[keyIndex_ % keys_.size()];
    }

    void upgradeAllFunctions(RagePhysicsEuphoria& phys, const std::string& reason = "AI RAGE Boost") {
        upgradeTier_ += 1;
        phys.spacebarClimbRate = std::min(55.0, phys.spacebarClimbRate + 2.5);
        phys.helicopterLiftAccel = std::min(65.0, phys.helicopterLiftAccel + 3.0);
        phys.parachuteDescentSpeed = std::max(2.5, phys.parachuteDescentSpeed - 0.35);
        phys.moonGravityScale = std::max(0.14, phys.moonGravityScale - 0.02);
        phys.euphoriaImpulseScale = std::min(2.5, phys.euphoriaImpulseScale + 0.12);
        lastUpgradeNote_ = "Tier #" + std::to_string(upgradeTier_) + ": " + reason;
    }

    std::string buildTuningJson(const RagePhysicsEuphoria& phys, const MemoryArena64& mem) const {
        std::ostringstream oss;
        oss << "{"
            << "\"version\":\"" << escapeJson(kEngineVersion) << "\","
            << "\"seed\":\"" << escapeJson(seed_) << "\","
            << "\"upgradeTier\":" << upgradeTier_ << ","
            << "\"lastUpgradeNote\":\"" << escapeJson(lastUpgradeNote_) << "\","
            << "\"groqKeysCount\":" << keys_.size() << ","
            << "\"memory64\":{"
            << "\"sector\":\"" << escapeJson(mem.sectorKey()) << "\","
            << "\"activeBytes\":" << mem.activeBytes() << ","
            << "\"peakBytes\":" << mem.peakBytes() << ","
            << "\"flushes\":" << mem.flushesCount()
            << "},"
            << "\"rageEuphoria\":{"
            << "\"spacebarClimbRate\":" << phys.spacebarClimbRate << ","
            << "\"helicopterLiftAccel\":" << phys.helicopterLiftAccel << ","
            << "\"parachuteDescentSpeed\":" << phys.parachuteDescentSpeed << ","
            << "\"fallDamageThreshold\":" << phys.fallDamageThreshold << ","
            << "\"fallDamageMultiplier\":" << phys.fallDamageMultiplier << ","
            << "\"moonGravityScale\":" << phys.moonGravityScale << ","
            << "\"euphoriaImpulseScale\":" << phys.euphoriaImpulseScale << ","
            << "\"waterCollisionBlockForCars\":true,"
            << "\"garageTotalVehicles\":400,"
            << "\"uniqueModelsCount\":54"
            << "}"
            << "}";
        return oss.str();
    }

    bool writeTuningJs(const std::string& path, const RagePhysicsEuphoria& phys, const MemoryArena64& mem) const {
        std::ofstream out(path, std::ios::binary);
        if (!out.is_open()) return false;
        out << "/**\n"
            << " * gta_cpp_tuning.js — Wygenerowane przez gta_rage.cpp\n"
            << " * Parametry silnika RAGE + Euphoria + 64-bit pamięci dla GTA SUS 3D: Infinity Edition.\n"
            << " */\n"
            << "(function (global) {\n"
            << "  'use strict';\n"
            << "  const GTA_CPP_TUNING = " << buildTuningJson(phys, mem) << ";\n"
            << "  global.GTA_CPP_TUNING = GTA_CPP_TUNING;\n"
            << "  global.GTA_SUS_CPP_RAGE = {\n"
            << "    tuning: GTA_CPP_TUNING,\n"
            << "    apply(state) {\n"
            << "      if (!state || typeof state !== 'object') return GTA_CPP_TUNING;\n"
            << "      state.cppRageTuning = Object.assign({}, state.cppRageTuning || {}, GTA_CPP_TUNING);\n"
            << "      return state.cppRageTuning;\n"
            << "    }\n"
            << "  };\n"
            << "})(typeof window !== 'undefined' ? window : globalThis);\n";
        return out.good();
    }

    /**
     * Wysyła zapytanie przez curl (jeśli dostępny w systemie) z rotacją kluczy Groq,
     * a gdy brak sieci lub curl — odpowiada lokalnie (w tym prostą matematykę np. 2+2).
     */
    std::string askGroqOrLocal(const std::string& prompt, const WorldActorState& state) {
        const std::string cleanPrompt = prompt.empty() ? "Co warto zrobić teraz w grze?" : prompt;
        const std::string sysMsg =
            "Jesteś asystentem AI silnika RAGE/Euphoria w GTA SUS 3D: Infinity Edition. "
            "Odpowiadasz po polsku na każde pytanie (doradzasz co robić w grze, liczysz matematykę np. 2+2=4, "
            "odpowiadasz na pytania ogólne). Stan gracza: HP=" + std::to_string(state.hp) +
            ", kasa=$" + std::to_string(state.cash) + ", gwiazdki=" + std::to_string(state.stars) + ".";

        const std::string payload =
            "{\"model\":\"" + std::string(kGroqModel) + "\","
            "\"temperature\":0.65,\"max_tokens\":400,"
            "\"messages\":["
            "{\"role\":\"system\",\"content\":\"" + escapeJson(sysMsg) + "\"},"
            "{\"role\":\"user\",\"content\":\"" + escapeJson(cleanPrompt) + "\"}"
            "]}";

        // Próba użycia curl jeśli środowisko pozwala na procesy zewnętrzne
        const char* disableNet = std::getenv("GTA_RAGE_OFFLINE");
        if (!disableNet || std::string(disableNet) != "1") {
            for (std::size_t attempt = 0; attempt < keys_.size(); ++attempt) {
                const std::size_t idx = (keyIndex_ + attempt) % keys_.size();
                std::string cmd =
                    "curl -sS -m 6 -X POST \"" + std::string(kGroqChatUrl) + "\" "
                    "-H \"Authorization: Bearer " + keys_[idx] + "\" "
                    "-H \"Content-Type: application/json\" "
                    "-d '" + payload + "' 2>/dev/null";
                FILE* pipe = popen(cmd.c_str(), "r");
                if (!pipe) break;
                std::string rawResp;
                char buf[512];
                while (std::fgets(buf, sizeof(buf), pipe) != nullptr) {
                    rawResp += buf;
                }
                int rc = pclose(pipe);
                if (rc == 0 && !rawResp.empty()) {
                    const std::string marker = "\"content\":\"";
                    std::size_t pos = rawResp.rfind(marker);
                    if (pos != std::string::npos) {
                        pos += marker.size();
                        std::string extracted;
                        for (std::size_t i = pos; i < rawResp.size(); ++i) {
                            if (rawResp[i] == '\\' && i + 1 < rawResp.size()) {
                                char nx = rawResp[++i];
                                if (nx == 'n') extracted += '\n';
                                else extracted += nx;
                            } else if (rawResp[i] == '"') {
                                break;
                            } else {
                                extracted += rawResp[i];
                            }
                        }
                        if (!extracted.empty()) {
                            keyIndex_ = idx;
                            return extracted;
                        }
                    }
                }
            }
        }

        // Lokalny fallback (np. 2+2 oraz wskazówki do gry)
        std::string lower = toLower(cleanPrompt);
        if (lower.find("2+2") != std::string::npos || lower.find("2 + 2") != std::string::npos) {
            return "2 + 2 = 4. W grze możesz teraz użyć Spacji do wznoszenia samolotu/helikoptera, odwiedzić rakietę NASA albo zrobić napad [J]/[H].";
        }
        if (state.onMoon) {
            return "Na Księżycu działa niska grawitacja — skacz wysoko i użyj [E] przy rakiecie, aby wrócić na Ziemię.";
        }
        if (state.stars >= 4) {
            return "Wysoki pościg (" + std::to_string(state.stars) + "★) — użyj nitro [N], mostów nad zatoką lub wnętrz budynków [E].";
        }
        return "Tryb RAGE/Euphoria aktywny (HP: " + std::to_string(state.hp) + ", kasa: $" + std::to_string(state.cash) +
               "). Trzymaj Spację w maszynach latających, unikaj wjazdu autami do wody i korzystaj ze spadochronu przy opadaniu.";
    }

private:
    std::string seed_;
    std::vector<std::string> keys_;
    std::size_t keyIndex_ = 0;
    int upgradeTier_ = 1;
    std::string lastUpgradeNote_ = "Tier #1: Domyślny profil RAGE + Euphoria 64-bit";
};

} // namespace gta_rage

int main(int argc, char** argv) {
    using namespace gta_rage;

    std::string seed = "DUBAI-SUS-64";
    bool selfTest = false;
    bool noServer = false;

    for (int i = 1; i < argc; ++i) {
        std::string arg = argv[i];
        if (arg == "--seed" && i + 1 < argc) {
            seed = argv[++i];
        } else if (arg == "--self-test") {
            selfTest = true;
        } else if (arg == "--no-server") {
            noServer = true;
        }
    }

    MemoryArena64 memory(1024.0);
    memory.setOrigin(0.0, 0.0, 0.0);
    memory.trackAllocation(64ULL * 1024ULL * 1024ULL);

    RagePhysicsEuphoria physics;
    GroqRageDirector director(seed);

    if (selfTest) {
        WorldActorState testState;
        testState.inVehicle = true;
        testState.vehicleType = "plane";
        testState.altitude = 10.0;
        physics.step(testState, 0.05, true, false);
        if (testState.vy <= 0.0) {
            std::cerr << "[BLAD] Wznoszenie samolotu na Spacji nie zadziałało.\n";
            return 1;
        }

        WorldActorState waterCar;
        waterCar.inVehicle = true;
        waterCar.vehicleType = "car";
        waterCar.inWater = true;
        physics.step(waterCar, 0.05, false, false);
        if (!waterCar.blockCarInWater) {
            std::cerr << "[BLAD] Blokada wjazdu auta do wody nie zadziałała.\n";
            return 1;
        }

        director.upgradeAllFunctions(physics, "Test wzmocnienia AI");
        std::string ans = director.askGroqOrLocal("ile to 2+2?", testState);
        if (ans.find("4") == std::string::npos) {
            std::cerr << "[BLAD] Odpowiedź matematyczna nie zawiera 4.\n";
            return 1;
        }

        memory.flushStreamPool();
        std::cout << "[SELF-TEST OK] gta_rage.cpp zweryfikowany (kluczy Groq: " << director.keyCount()
                  << ", sektor: " << memory.sectorKey() << ").\n";
        return 0;
    }

    director.upgradeAllFunctions(physics, "Start silnika RAGE + Euphoria 64-bit");
    const std::string outJs = "gta_cpp_tuning.js";
    if (!director.writeTuningJs(outJs, physics, memory)) {
        std::cerr << "[UWAGA] Nie udało się zapisać " << outJs << "\n";
        return 1;
    }

    std::cout << "[OK] Wygenerowano " << outJs << " (seed: " << seed
              << ", kluczy Groq: " << director.keyCount()
              << ", sektor 64-bit: " << memory.sectorKey() << ")\n";
    if (noServer) {
        std::cout << "[INFO] Tryb --no-server: plik konfiguracyjny gotowy do użycia z index.html, gta.py i Gta.java.\n";
    }
    return 0;
}
