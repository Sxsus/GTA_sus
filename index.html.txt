  function seedToNumber(seedStr) {
    const s = String(seedStr || 'DUBAI-NYC-INFINITY-2026');
    let h = 2166136261 >>> 0;
    for (let i = 0; i < s.length; i++) {
      h ^= s.charCodeAt(i);

    }
    return h || 1337420;
  }

  function createRng(seedStr) {
    let a = seedToNumber(seedStr) >>> 0;
    return function nextFloat() {
      a |= 0;
      a = (a + 0x6D2B79F5) | 0;
      let t = Math.imul(a ^ (a >>> 15), 1 | a);
      t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
      return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    };
  }

  function evaluateOfflineMath(question) {
    const cleaned = question
      .toLowerCase()
      .replace(/ile to jest|ile to|policz|oblicz|wynik|równa się|wyjaśnij krótko dodawanie|\?/gi, '')
      .replace(/razy|x/gi, '*')
      .replace(/podzielić przez|przez|:/gi, '/')
      .replace(/plus/gi, '+')
      .replace(/minus/gi, '-')
      .replace(/,/g, '.')
      .trim();

    if (!/^[0-9+\-*/().\s%]+$/.test(cleaned)) return null;
    if (!/[0-9]/.test(cleaned) || !/[+\-*/%]/.test(cleaned)) return null;

    try {
      const expr = cleaned.replace(/(\d+(?:\.\d+)?)%/g, '($1/100)');
      const result = Function(`"use strict"; return (${expr});`)();
      if (typeof result === 'number' && Number.isFinite(result)) {
        const formatted = Number.isInteger(result) ? String(result) : String(Number(result.toFixed(4)));
        return `${cleaned.trim()} = **${formatted}**`;
      }
    } catch (_e) {}
    return null;
  }

  const BuiltinOptimizer = {
    drsScale: 1.0,
    smoothFrameMs: 16.7,
    fps: 60,
    culledCount: 0,
    lastAiTime: 0,
    aiBusy: false,
    update(dtSec, playerPos) {
      const dtMs = Math.max(1, Math.min(250, dtSec * 1000));
      this.smoothFrameMs = this.smoothFrameMs * 0.9 + dtMs * 0.1;
      this.fps = Math.max(1, Math.round(1000 / this.smoothFrameMs));

      let nextScale = this.drsScale;
      if (this.smoothFrameMs > 23.0) {
        nextScale = Math.max(0.65, Number((nextScale - 0.03).toFixed(2)));
      } else if (this.smoothFrameMs < 14.5 && nextScale < 1.0) {
        nextScale = Math.min(1.0, Number((nextScale + 0.02).toFixed(2)));
      }

      if (nextScale !== this.drsScale) {
        this.drsScale = nextScale;
        if (renderer && typeof renderer.setPixelRatio === 'function') {
          const baseDpr = Math.min(window.devicePixelRatio || 1, 2);
          renderer.setPixelRatio(Math.max(0.6, baseDpr * this.drsScale));
        }
      }

      let culled = 0;
      const cullDistSq = 310 * 310;
      if (playerPos) {
        for (let i = 0; i < npcs.length; i++) {
          const n = npcs[i];
          if (!n || !n.mesh) continue;
          const dx = n.x - playerPos.x;
          const dz = n.z - playerPos.z;
          const vis = (dx * dx + dz * dz) <= cullDistSq;
          n.mesh.visible = vis;
          if (!vis) culled++;
        }
        for (let i = 0; i < vehicles.length; i++) {
          const v = vehicles[i];
          if (!v || !v.mesh) continue;
          const dx = v.x - playerPos.x;
          const dz = v.z - playerPos.z;
          const vis = v.id === state.inVehicleId || (dx * dx + dz * dz) <= (cullDistSq * 1.25);
          v.mesh.visible = vis;
          if (!vis) culled++;
        }
      }
      this.culledCount = culled;

      const extOpt = window.GTA_SUS_OPTIMIZER || window.optymalizer;
      if (extOpt && typeof extOpt.update === 'function') {
        try {
          if (extOpt.update.length >= 2 && typeof extOpt.cullCollections === 'function') {
            extOpt.update(performance.now(), playerPos);
          } else {
            extOpt.update(dtSec, { player: playerPos, npcs, vehicles, clouds, particles });
          }
        } catch (_e) {}
      }

      const gfx = getExternalGrafika();
      if (gfx && typeof gfx.update === 'function') {
        try {
          gfx.update(dtSec, {
            renderer,
            scene,
            camera,
            dirLight,
            hemiLight,
            player: playerPos,
            vehicles,
            npcs,
            clouds,
            state
          });
        } catch (_e) {}
      }

      if ($('hudOptStats')) {
        $('hudOptStats').textContent = `FPS: ${this.fps} • DRS ${Math.round(this.drsScale * 100)}% • Cull ${this.culledCount}`;
      }
    },
    async tuneWithAi(force = false) {
      const now = Date.now();
      if (!force && now - this.lastAiTime < 12000) return;
      if (this.aiBusy) return;
      this.aiBusy = true;
      this.lastAiTime = now;
      try {
        const raw = await queryGroqWithRotation(
          [
            {
              role: 'system',
              content:
                'You are a 3D graphics & FPS tuning advisor for GTA SUS 3D. Return ONLY JSON with keys: exposure (0.7..1.45), cloudDensity (10..100), scale (0.65..1.0).'
            },
            {
              role: 'user',
              content: `FPS=${this.fps}, preset=${state.qualityPreset}, lighting=${state.lightingMode}, exposure=${state.exposure}, clouds=${state.cloudDensity}`
            }
          ],
          {
            primaryKey: GROQ_KEYS.optimizer,
            maxTokens: 110,
            temperature: 0.25,
            timeoutMs: 3200,
            jsonMode: true
          }
        );
        const cleaned = raw.replace(/```json|```/gi, '').trim();
        const parsed = JSON.parse(cleaned);
        if (typeof parsed.exposure === 'number') state.exposure = Math.max(0.65, Math.min(1.55, parsed.exposure));
        if (typeof parsed.cloudDensity === 'number') state.cloudDensity = Math.max(0, Math.min(100, Math.round(parsed.cloudDensity)));
        if (typeof parsed.scale === 'number') this.drsScale = Math.max(0.65, Math.min(1.0, parsed.scale));
      } catch (_err) {
        if (this.fps < 35) {
          this.drsScale = 0.75;
          state.cloudDensity = Math.max(25, state.cloudDensity - 20);
        } else {
          this.drsScale = 1.0;

      } finally {

  function disposeObject3D(obj) {
    if (!obj) return;
    if (obj.parent) obj.parent.remove(obj);
    obj.traverse((child) => {
      if (child.geometry && typeof child.geometry.dispose === 'function') {
        child.geometry.dispose();
      }
      if (child.material) {
        const mats = Array.isArray(child.material) ? child.material : [child.material];
        mats.forEach((m) => {
          if (m && typeof m.dispose === 'function') m.dispose();
        });
      }
    });
  }

  function clearWorldObjects() {

  function makeSkyscraperWindowTexture(mode = 'day', seedVal = 1) {
    const gfx = getExternalGrafika();
    if (gfx && typeof gfx.createWindowTexture === 'function') {
      try {
        const customTex = gfx.createWindowTexture(mode, seedVal, THREE);
        if (customTex) return customTex;
      } catch (_e) {}
    }

    const canvas = document.createElement('canvas');
    canvas.width = 128;
    canvas.height = 256;
    const ctx = canvas.getContext('2d');
    const rng = createRng(`WIN-${seedVal}-${mode}`);

    ctx.fillStyle = mode === 'night' ? '#0f172a' : '#1e293b';
    ctx.fillRect(0, 0, 128, 256);

    const cols = 8;
    const rows = 24;
    const padX = 5;
    const padY = 4;
    const cellW = (128 - padX * (cols + 1)) / cols;
    const cellH = (256 - padY * (rows + 1)) / rows;

    for (let r = 0; r < rows; r++) {
      for (let c = 0; c < cols; c++) {
        const litProb = mode === 'night' ? 0.68 : (mode === 'sunset' ? 0.42 : 0.18);
        const lit = rng() < litProb;
        if (lit) {
          ctx.fillStyle = mode === 'night' ? (rng() > 0.25 ? '#fef08a' : '#7dd3fc') : '#fde68a';
        } else {
          ctx.fillStyle = mode === 'night' ? '#172554' : '#38bdf8';
        }
        ctx.fillRect(
          padX + c * (cellW + padX),
          padY + r * (cellH + padY),
          cellW,
          cellH
        );
      }
    }

    const tex = new THREE.CanvasTexture(canvas);
    tex.wrapS = THREE.RepeatWrapping;
    tex.wrapT = THREE.RepeatWrapping;
    tex.repeat.set(1, 2);
    return tex;
  }

    renderer.shadowMap.enabled = true;
    renderer.shadowMap.type = THREE.PCFSoftShadowMap;
    renderer.toneMapping = THREE.ACESFilmicToneMapping;
    renderer.toneMappingExposure = state.exposure;

    scene = new THREE.Scene();
    scene.background = new THREE.Color(0x5da9e9);
    scene.fog = new THREE.FogExp2(0x7ec0ee, 0.00095);

    hemiLight = new THREE.HemisphereLight(0xdbeafe, 0x334155, 0.9);
    scene.add(hemiLight);

    dirLight = new THREE.DirectionalLight(0xfffbeb, 1.3);
    dirLight.position.set(110, 165, 85);
    dirLight.castShadow = true;
    dirLight.shadow.mapSize.set(2048, 2048);

  function handleResize() {
    if (!renderer || !camera) return;

    const h = Math.max(1, window.innerHeight);
    camera.aspect = w / h;
    camera.updateProjectionMatrix();
    renderer.setSize(w, h);
    syncGrafikaHooks('resize');
  }

  function applyLightingAndGraphics() {
    if (!renderer || !scene) return;
    renderer.toneMapping = THREE.ACESFilmicToneMapping;
    renderer.toneMappingExposure = state.exposure;

    const presetMap = {
      low: { shadowRes: 512, dprMax: 1.0, shadows: false },
      medium: { shadowRes: 1024, dprMax: 1.35, shadows: true },
      high: { shadowRes: 2048, dprMax: 1.75, shadows: true },
      ultra: { shadowRes: 4096, dprMax: 2.0, shadows: true }
    };
    const q = presetMap[state.qualityPreset] || presetMap.high;
    renderer.shadowMap.enabled = q.shadows;
    renderer.setPixelRatio(Math.min(window.devicePixelRatio || 1, q.dprMax) * BuiltinOptimizer.drsScale);

    if (dirLight) {
      dirLight.castShadow = q.shadows;
      if (dirLight.shadow.mapSize.width !== q.shadowRes) {
        dirLight.shadow.mapSize.set(q.shadowRes, q.shadowRes);
        if (dirLight.shadow.map) {
          dirLight.shadow.map.dispose();
          dirLight.shadow.map = null;
        }
      }
    }

    if (state.onMoon) {
      scene.background = new THREE.Color(0x030712);
      if (scene.fog) scene.fog.color.setHex(0x050a18);
      if (hemiLight) { hemiLight.color.setHex(0x94a3b8); hemiLight.groundColor.setHex(0x1e293b); hemiLight.intensity = 0.55; }
      if (dirLight) { dirLight.color.setHex(0xf8fafc); dirLight.intensity = 1.45; dirLight.position.set(120, 140, 80); }
    } else if (state.lightingMode === 'night') {
      scene.background = new THREE.Color(0x070d1f);
      if (scene.fog) scene.fog.color.setHex(0x0b132b);
      if (hemiLight) { hemiLight.color.setHex(0x3b82f6); hemiLight.groundColor.setHex(0x0f172a); hemiLight.intensity = 0.48; }
      if (dirLight) { dirLight.color.setHex(0x93c5fd); dirLight.intensity = 0.58; dirLight.position.set(-90, 120, -70); }
    } else if (state.lightingMode === 'sunset') {
      scene.background = new THREE.Color(0xf97316);
      if (scene.fog) scene.fog.color.setHex(0xfb923c);
      if (hemiLight) { hemiLight.color.setHex(0xfde68a); hemiLight.groundColor.setHex(0x451a03); hemiLight.intensity = 0.85; }
      if (dirLight) { dirLight.color.setHex(0xffedd5); dirLight.intensity = 1.38; dirLight.position.set(130, 46, -85); }
    } else {
      scene.background = new THREE.Color(0x5da9e9);
      if (scene.fog) scene.fog.color.setHex(0x8ecae6);
      if (hemiLight) { hemiLight.color.setHex(0xe0f2fe); hemiLight.groundColor.setHex(0x334155); hemiLight.intensity = 0.92; }
      if (dirLight) { dirLight.color.setHex(0xfffbeb); dirLight.intensity = 1.3; dirLight.position.set(110, 165, 85); }
    }

    clouds.forEach((c, i) => {
      const visibleCount = state.onMoon ? 0 : Math.round((state.cloudDensity / 100) * clouds.length);
      c.mesh.visible = i < visibleCount;
      c.mesh.traverse((child) => {
        if (child.isMesh && child.material) {
          child.material.color.setHex(
            state.lightingMode === 'night' ? 0x1e293b : state.lightingMode === 'sunset' ? 0xfed7aa : 0xffffff
          );
        }
      });
    });

    if ($('setExposureVal')) $('setExposureVal').textContent = state.exposure.toFixed(2);
    if ($('setCloudDensityVal')) $('setCloudDensityVal').textContent = `${state.cloudDensity}%`;
    syncGrafikaHooks('graphicsChanged');
  }

  function triggerRocketFlight() {
    if (state.rocketPhase !== 'none') return;
    if (state.onMoon) {
      state.inRocket = true;
      state.rocketPhase = 'return';
      state.rocketTimer = 2.4;
      if (rocketPlumeMesh) rocketPlumeMesh.visible = true;
      SoundFX.rocketLaunch();
      showToast('🚀 Odpalono silniki powrotne z Księżyca na Ziemię!');
      return;
    }

    const distPad = Math.hypot(state.player.x - rocketPadPos.x, state.player.z - rocketPadPos.z);
    if (distPad > 22) {
      state.player.x = rocketPadPos.x;
      state.player.y = 0;
      state.player.z = rocketPadPos.z + 10;
    }
    state.inRocket = true;
    state.rocketPhase = 'launch';
    state.rocketTimer = 2.6;
    if (rocketPlumeMesh) rocketPlumeMesh.visible = true;
    SoundFX.rocketLaunch();
    addLog('Start rakiety NASA w kierunku Księżyca!', 'text-orange-300');
    showToast('🚀 Start rakiety NASA na Księżyc!');
  }

  function isDeepWaterAt(x, z) {
    for (let i = 0; i < waterZones.length; i++) {
      const wz = waterZones[i];
      if (x >= wz.minX && x <= wz.maxX && z >= wz.minZ && z <= wz.maxZ) {
        return true;
      }
    }
    return false;
  }

    );

  async function generateSeedWorldWithAi(seedInputValue, sourceLabel = 'start') {
    const seed = String(seedInputValue || 'DUBAI-NYC-INFINITY-2026').trim() || 'DUBAI-NYC-INFINITY-2026';

    const statusEl = sourceLabel === 'start' ? $('startSeedStatus') : null;
    if (statusEl) statusEl.textContent = 'Generowanie AI / Seed...';

    let aiPlan = null;
    try {
      const raw = await queryGroqWithRotation(
        [
          {
            role: 'system',
            content:
              'You generate a JSON world theme for a massive Dubai/US metropolis in GTA SUS 3D from a seed. Return ONLY JSON with keys: theme ("day"|"sunset"|"night"), towerBoost (boolean), note (short Polish phrase).'
          },
          {
            role: 'user',
            content: `Seed: ${seed}`
          }
        ],
        {
          primaryKey: GROQ_KEYS.worldSeed[0],
          pool: GROQ_KEYS.worldSeed,
          maxTokens: 120,
          temperature: 0.4,
          timeoutMs: 3400,
          jsonMode: true
        }
      );
      const cleaned = raw.replace(/```json|```/gi, '').trim();
      aiPlan = JSON.parse(cleaned);
      if (statusEl) statusEl.textContent = `AI Seed: ${aiPlan.note || 'Gotowe'}`;
    } catch (_err) {
      const rng = createRng(seed);
      const themes = ['day', 'sunset', 'night'];
      aiPlan = {
        theme: themes[Math.floor(rng() * themes.length)],
        towerBoost: rng() > 0.4,
        note: 'Deterministyczny seed PRNG (offline)'
      };
      if (statusEl) statusEl.textContent = aiPlan.note;
    }

    buildWorldFromSeed(seed, aiPlan);
    addLog(`Wygenerowano metropolię z seedu "${seed}" (${aiPlan.theme})`, 'text-cyan-300');
    showToast(`Świat wygenerowany dla seedu: ${seed}`);
  }

  function applyChaosEvent(ev) {
    const now = performance.now();
    state.liveEvent = {
      title: ev.title || '⚡ Zdarzenie AI Chaos',
      desc: ev.desc || 'Metropolia reaguje na impuls dyrektora AI!',
      endsAt: now + (ev.durationSec || 28) * 1000,
      type: ev.type || 'bonus'
    };

    if (ev.type === 'cash_surge') {
      state.cash += 600;
      SoundFX.pickup();
    } else if (ev.type === 'police_raid') {
      state.wantedLevel = Math.min(6, state.wantedLevel + 2);
      for (let i = 0; i < 3; i++) {
        spawnNpc('swat', state.player.x + (i - 1) * 8, state.player.z - 16, 0);
      }
      SoundFX.tone(420, 0.25, 'sawtooth', 0.08);
    } else if (ev.type === 'air_drop') {
      spawnSupplyDrop();
    } else if (ev.type === 'meteor_shock') {
      SoundFX.explosion();
      npcs.slice(0, 5).forEach((n) => {
        triggerRagdoll(n, (Math.random() - 0.5) * 14, 7, (Math.random() - 0.5) * 14, 2.8);
      });
    } else if (ev.type === 'fog_storm') {
      if (scene && scene.fog) {
        scene.fog.density = 0.0036;
        setTimeout(() => {
          if (scene && scene.fog) scene.fog.density = 0.00095;
        }, 20000);
      }
    }

    const banner = $('hudLiveEventBanner');
    if (banner) {
      banner.classList.remove('hidden');
      $('liveEventTitle').textContent = state.liveEvent.title;
      $('liveEventDesc').textContent = state.liveEvent.desc;
    }
    if ($('phoneChaosInfo')) {

    }
    addLog(`${state.liveEvent.title} — ${state.liveEvent.desc}`, 'text-fuchsia-300');
    updateHudLabels();
  }

  function findNearestNpc(maxDist = 6.5) {
    let best = null;
    let bestD = maxDist;
    for (let i = 0; i < npcs.length; i++) {
      const n = npcs[i];
      if (!n || !n.mesh || !n.mesh.visible) continue;
      const d = Math.hypot(n.x - state.player.x, n.z - state.player.z);
      if (d < bestD) {
        bestD = d;
        best = n;
      }
    }
    return best;
  }

      const d = Math.hypot(v.x - state.player.x, v.z - state.player.z);
      if (d < bestD) {
        bestD = d;
        best = v;
      }
    }
    return best;
  }

  function findNearestDoor(maxDist = 4.5) {
    let best = null;
    let bestD = maxDist;
    for (let i = 0; i < buildings.length; i++) {
      const b = buildings[i];
      const d = Math.hypot(b.doorX - state.player.x, b.doorZ - state.player.z);
      if (d < bestD) {
        bestD = d;
        best = b;
      }
    }
    return best;
  }

  function findNearestRobbableShop(maxDist = 5.5) {
    let best = null;
    let bestD = maxDist;
    for (let i = 0; i < robbableShops.length; i++) {
      const s = robbableShops[i];
      const d = Math.hypot(s.robX - state.player.x, s.robZ - state.player.z);
      if (d < bestD) {
        bestD = d;
        best = s;
      }
    }
    return best;
  }

  function robNearbyShop() {
    const shop = findNearestRobbableShop(7.0);
    if (!shop) {
      showToast('Podejdź bliżej sklepu do okradania (np. użyj GPS w SusPhone [P])');
      return;
    }
    if (shop.cooldown > 0) {
      showToast(`Kasa w "${shop.name}" jest chwilowo pusta (${Math.ceil(shop.cooldown)}s)`);
      return;
    }
    shop.cooldown = 35;
    state.cash += shop.reward;
    state.wantedLevel = Math.min(6, Math.max(state.wantedLevel, 1) + 1);
    SoundFX.pickup();
    addLog(`Okradziono "${shop.name}"! Łup: +$${shop.reward} (Pościg: ${state.wantedLevel}★)`, 'text-rose-300');
    showToast(`💰 Łup ze sklepu: +$${shop.reward}!`);
    updateHudLabels();
  }

  function deployParachute() {
    if (state.inVehicleId || state.onMoon || state.player.grounded || state.player.y <= 3.0) {
      return false;
    }
    if (state.parachuteOpen) return true;
    if (state.parachutes <= 0) {
      showToast('Brak spadochronów! Kup za $100 w menu [I]');
      return false;
    }
    state.parachutes--;
    state.parachuteOpen = true;
    state.player.maxFallSpeed = 0;
    if (parachuteMesh) parachuteMesh.visible = true;
    SoundFX.tone(360, 0.16, 'triangle', 0.06);
    addLog(`Otwarto spadochron! (Pozostało: ${state.parachutes})`, 'text-sky-300');
    showToast('🪂 Spadochron otwarty!');
    return true;
  }

  function getContextualPrompt() {
    if (state.insideBuildingId) {
      const b = buildings.find((item) => item.id === state.insideBuildingId);
      return `[E] Wyjdź z wnętrza: ${b ? b.name : 'Budynek'}`;
    }
    if (state.onMoon) {
      return `[E] Wejdź do Rakiety NASA i wróć na Ziemię (Niska grawitacja na Księżycu!)`;
    }
    if (state.inVehicleId) {
      const v = vehicles.find((item) => item.id === state.inVehicleId);
      if (v && (v.kind === 'airplane' || v.kind === 'helicopter')) {
        return `[E/F] Wysiądź (uwaga na upadek — użyj [O/Spacji] dla spadochronu!) • Trzymaj [SPACJĘ] lub [R] aby wznosić`;
      }
      return `[E/F] Wysiądź z pojazdu: ${v ? v.name : 'Pojazd'} • [N] Nitro`;
    }

    const distRocket = Math.hypot(rocketPadPos.x - state.player.x, rocketPadPos.z - state.player.z);
    if (distRocket < 14) {
      return `[E] Wejdź do Rakiety NASA i leć na KSIĘŻYC!`;
    }

    const shop = findNearestRobbableShop(5.0);
    if (shop) {
      return `[J] Okradnij sklep: ${shop.name} (+$${shop.reward})`;
    }

    const distVault = Math.hypot(bankVault.x - state.player.x, bankVault.z - state.player.z);
    if (distVault < 4.5) {
      return `[H] Napad na skarbiec banku (Postęp: ${Math.round(bankVault.progress)}%)`;
    }

    const door = findNearestDoor(4.2);
    const veh = findNearestVehicle(5.8);
    const npc = findNearestNpc(4.8);

    const candidates = [];
    if (door) {
      candidates.push({
        dist: Math.hypot(door.doorX - state.player.x, door.doorZ - state.player.z),
        text: `[E] Wejdź do budynku: ${door.name}`
      });
    }
    if (veh) {
      candidates.push({
        dist: Math.hypot(veh.x - state.player.x, veh.z - state.player.z),
        text: `[E/F] ${veh.aiControlled ? 'Ukradnij i wsiądź' : 'Wsiądź'}: ${veh.name} ($0)`
      });
    }
    if (npc) {
      candidates.push({
        dist: Math.hypot(npc.x - state.player.x, npc.z - state.player.z),
        text: `[T] Porozmawiaj z AI: ${npc.name}`
      });
    }

    if (candidates.length === 0) return null;
    candidates.sort((a, b) => a.dist - b.dist);
    return candidates[0].text;
  }

  function enterOrExitVehicle(forceExit = false) {
    if (state.insideBuildingId || state.onMoon) return;
    if (state.inVehicleId) {
      const cur = vehicles.find((v) => v.id === state.inVehicleId);
      state.inVehicleId = null;
      if (cur) {
        state.player.x = cur.x + Math.cos(cur.yaw) * 3.2;
        state.player.z = cur.z - Math.sin(cur.yaw) * 3.2;
        state.player.y = Math.max(0, cur.y);
        state.player.vy = cur.y > 4 ? -3.0 : 0;
        state.player.grounded = cur.y <= 0.4;
        state.player.maxFallSpeed = Math.abs(state.player.vy);
      }
      addLog(`Wysiadłeś z ${cur ? cur.name : 'pojazdu'}.`);
      updateHudLabels();
      return;
    }
    if (forceExit) return;

    const veh = findNearestVehicle(6.2);
    if (!veh) return;
    if (veh.aiControlled) {
      veh.aiControlled = false;
      state.wantedLevel = Math.min(6, state.wantedLevel + 1);
      addLog(`Przejęto pojazd ${veh.name} (+1★ poszukiwań)!`, 'text-amber-300');
    } else {
      addLog(`Wsiadłeś do: ${veh.name}`, 'text-cyan-300');
    }
    state.inVehicleId = veh.id;
    state.parachuteOpen = false;
    if (parachuteMesh) parachuteMesh.visible = false;
    state.player.x = veh.x;
    state.player.y = veh.y + 1.2;
    state.player.z = veh.z;
    updateHudLabels();
  }

  function toggleBuildingInterior() {
    if (state.inVehicleId || state.onMoon) return;
    if (state.insideBuildingId) {
      const cur = buildings.find((b) => b.id === state.insideBuildingId);
      state.insideBuildingId = null;
      const out = cur ? cur.outsideSpawn : state.player.lastSavedOutside;
      state.player.x = out.x;
      state.player.y = 0;
      state.player.z = out.z;
      state.player.vx = 0;
      state.player.vz = 0;
      addLog(`Wyszedłeś z budynku: ${cur ? cur.name : 'Wnętrze'}`);
      updateHudLabels();
      return;
    }

    const door = findNearestDoor(4.4);
    if (!door) return;
    state.player.lastSavedOutside = { x: door.outsideSpawn.x, y: 0, z: door.outsideSpawn.z };
    state.insideBuildingId = door.id;
    state.player.x = door.interiorSpawn.x;
    state.player.y = 0;
    state.player.z = door.interiorSpawn.z;
    state.player.vx = 0;
    state.player.vz = 0;
    addLog(`Wszedłeś do wnętrza: ${door.name} ([E] aby wyjść)`, 'text-emerald-300');
    if (door.id === 'safehouse') {
      state.hp = Math.min(state.maxHp, state.hp + 35);
      addLog('Odpoczynek w Penthouse: +35 HP', 'text-emerald-400');
    }
    updateHudLabels();
  }

  function handleContextActionE() {
    if (state.onMoon) {
      triggerRocketFlight();
      return;
    }
    if (state.insideBuildingId) {
      toggleBuildingInterior();
      return;
    }
    if (state.inVehicleId) {
      enterOrExitVehicle(true);
      return;
    }

    const distRocket = Math.hypot(rocketPadPos.x - state.player.x, rocketPadPos.z - state.player.z);
    if (distRocket < 14) {
      triggerRocketFlight();
      return;
    }

    const door = findNearestDoor(4.2);
    const veh = findNearestVehicle(5.8);
    if (door && (!veh || Math.hypot(door.doorX - state.player.x, door.doorZ - state.player.z) <= Math.hypot(veh.x - state.player.x, veh.z - state.player.z))) {
      toggleBuildingInterior();
    } else if (veh) {
      enterOrExitVehicle(false);
    }
  }

  function performBankHeistStep() {

    if (dist > 5.5) {

    state.wantedLevel = Math.min(6, Math.max(state.wantedLevel, 2));

    } else {

  }

  function firePlayerWeapon() {
    const w = state.weapons[state.activeSlot] || state.weapons[1];
    if (state.activeSlot !== 1) {
      if (w.ammo <= 0) {
        showToast('Brak amunicji! Kup w sklepie [I] za $100');
        return;
      }
      w.ammo--;
    }

    const isHeavy = state.activeSlot === 5;
    SoundFX.shot(isHeavy);

    const speed = isHeavy ? 56 : 74;
    const dirX = Math.sin(state.player.yaw);
    const dirZ = Math.cos(state.player.yaw);
    const startX = state.player.x + dirX * 1.4;
    const startY = state.player.y + 1.5;
    const startZ = state.player.z + dirZ * 1.4;

    const projMesh = new THREE.Mesh(
      new THREE.SphereGeometry(isHeavy ? 0.36 : 0.16, 8, 8),
      new THREE.MeshBasicMaterial({ color: isHeavy ? 0xf97316 : 0xfef08a })
    );
    projMesh.position.set(startX, startY, startZ);
    scene.add(projMesh);

    projectiles.push({
      mesh: projMesh,
      vx: dirX * speed,
      vy: 0.3,
      vz: dirZ * speed,
      life: 1.4,
      explosive: isHeavy,
      dmg: w.dmg,
      fromPlayer: true
    });
    updateHudLabels();
  }

  function spawnExplosion(x, y, z, radius = 9) {
    SoundFX.explosion();
    for (let i = 0; i < 16; i++) {
      const pMesh = new THREE.Mesh(
        new THREE.SphereGeometry(0.45, 6, 6),
        new THREE.MeshBasicMaterial({ color: i % 2 === 0 ? 0xf97316 : 0xfacc15 })
      );
      pMesh.position.set(x, Math.max(0.4, y), z);
      scene.add(pMesh);
      particles.push({
        mesh: pMesh,
        vx: (Math.random() - 0.5) * 18,
        vy: 3 + Math.random() * 11,
        vz: (Math.random() - 0.5) * 18,
        life: 0.65 + Math.random() * 0.35
      });
    }

    npcs.forEach((n) => {
      const dx = n.x - x;
      const dz = n.z - z;
      const dist = Math.hypot(dx, dz);
      if (dist < radius) {
        const falloff = Math.max(0.25, 1 - dist / radius);
        triggerRagdoll(n, (dx / Math.max(0.1, dist)) * 14 * falloff, 8 * falloff, (dz / Math.max(0.1, dist)) * 14 * falloff, 2.6);
        n.hp -= 85 * falloff;
      }
    });
  }

  function updatePhysics(dt) {
    const safeDt = Math.min(0.05, Math.max(0.001, dt));
    const now = performance.now();

    for (let i = 0; i < robbableShops.length; i++) {
      if (robbableShops[i].cooldown > 0) {
        robbableShops[i].cooldown = Math.max(0, robbableShops[i].cooldown - safeDt);
      }
    }

    if (state.liveEvent) {
      const remSec = Math.max(0, Math.ceil((state.liveEvent.endsAt - now) / 1000));
      if ($('liveEventTimer')) $('liveEventTimer').textContent = `${remSec}s`;
      if (remSec <= 0) {
        state.liveEvent = null;
        if ($('hudLiveEventBanner')) $('hudLiveEventBanner').classList.add('hidden');
      }
    }

    if (state.wantedLevel > 0) {
      state.wantedDecayTimer += safeDt;
      if (state.wantedDecayTimer > 18) {
        state.wantedDecayTimer = 0;
        state.wantedLevel = Math.max(0, state.wantedLevel - 1);
        updateHudLabels();
      }
    }

    if (state.rocketPhase !== 'none' && rocketGroup) {
      state.rocketTimer -= safeDt;
      const dir = state.rocketPhase === 'launch' ? 1 : -1;
      rocketGroup.position.y += dir * 32 * safeDt;
      if (rocketPlumeMesh) {
        rocketPlumeMesh.visible = true;
        rocketPlumeMesh.scale.y = 0.85 + Math.sin(now * 0.04) * 0.3;
      }
      state.player.x = rocketGroup.position.x;
      state.player.y = rocketGroup.position.y + 8;
      state.player.z = rocketGroup.position.z;
      camera.position.set(rocketGroup.position.x - 16, rocketGroup.position.y + 16, rocketGroup.position.z + 20);
      camera.lookAt(rocketGroup.position.x, rocketGroup.position.y + 12, rocketGroup.position.z);
      renderer.render(scene, camera);
      if (state.rocketTimer <= 0) {
        if (state.rocketPhase === 'launch') switchToMoonSurface();
        else returnFromMoonToEarth();
      }
      return;
    }

    if (fountainWaterMesh && !state.onMoon) {
      fountainWaterMesh.scale.y = 1 + Math.sin(now * 0.006) * 0.18;
    }

    for (let i = 0; i < clouds.length; i++) {
      const c = clouds[i];
      if (!c.mesh.visible) continue;
      c.mesh.position.x += c.speed * safeDt;
      c.mesh.position.y = c.baseY + Math.sin(now * 0.0008 + c.phase) * 0.8;
      if (c.mesh.position.x > 310) c.mesh.position.x = -310;
    }

    if (keys['ArrowLeft']) state.player.camYaw += 2.1 * safeDt;
    if (keys['ArrowRight']) state.player.camYaw -= 2.1 * safeDt;
    if (keys['ArrowUp']) state.player.camPitch = Math.min(0.95, state.player.camPitch + 1.4 * safeDt);
    if (keys['ArrowDown']) state.player.camPitch = Math.max(-0.35, state.player.camPitch - 1.4 * safeDt);

    const activeVeh = state.inVehicleId ? vehicles.find((v) => v.id === state.inVehicleId) : null;

    if (activeVeh && !state.onMoon) {
      const v = activeVeh;
      const forwardInput = (keys['KeyW'] ? 1 : 0) - (keys['KeyS'] ? 1 : 0);
      const turnInput = (keys['KeyA'] ? 1 : 0) - (keys['KeyD'] ? 1 : 0);
      const spaceHeld = Boolean(keys['Space']);
      const climbHeld = spaceHeld || Boolean(keys['KeyR']);
      const descendHeld = Boolean(keys['KeyQ']) || Boolean(keys['ShiftLeft']) || Boolean(keys['ShiftRight']);
      const nitroHeld = Boolean(keys['KeyN']) && v.hasNitro && v.nitro > 1;

      if (nitroHeld) {
        v.nitro = Math.max(0, v.nitro - 28 * safeDt);
      } else {
        v.nitro = Math.min(100, v.nitro + 12 * safeDt);
      }
      const boostMult = nitroHeld ? 1.55 : 1.0;

      if (v.kind === 'airplane') {
        if (v.propeller) v.propeller.rotation.z += (28 + Math.abs(v.speed) * 1.4) * safeDt;

        const thrustBoost = spaceHeld ? 1.45 : 1.0;
        if (forwardInput > 0) {
          v.speed = Math.min(v.maxSpeed * 1.18, v.speed + v.accel * thrustBoost * safeDt);
        } else if (forwardInput < 0) {
          v.speed = Math.max(-10, v.speed - v.accel * 0.9 * safeDt);
        } else if (spaceHeld) {
          v.speed = Math.min(v.maxSpeed, v.speed + v.accel * 0.72 * safeDt);
        } else {
          v.speed *= Math.pow(0.84, safeDt);
        }

        const speedFactor = Math.max(0.35, Math.min(1.0, Math.abs(v.speed) / 18));
        v.yaw += turnInput * (v.grounded ? 1.45 : 1.85) * speedFactor * safeDt;

        if (climbHeld) {
          v.vy = Math.min(20, v.vy + (spaceHeld ? 30 : 23) * safeDt);
          v.pitch = Math.min(0.45, v.pitch + 1.2 * safeDt);
          if (Math.abs(v.speed) < 12) v.speed = Math.min(v.maxSpeed, v.speed + 16 * safeDt);
        } else if (descendHeld && v.y > 0.1) {
          v.vy = Math.max(-14, v.vy - 20 * safeDt);
          v.pitch = Math.max(-0.38, v.pitch - 1.0 * safeDt);
        } else if (v.y > 0.2) {
          const liftBalance = Math.min(1.0, Math.abs(v.speed) / 32);
          const targetVy = -6.8 * (1 - liftBalance * 0.7);
          v.vy += (targetVy - v.vy) * 3.0 * safeDt;
          v.pitch += (0 - v.pitch) * 2.5 * safeDt;
        } else {
          v.vy = Math.min(0, v.vy);
          v.pitch += (0 - v.pitch) * 5 * safeDt;
        }

        const horizSpeed = v.speed * Math.cos(v.pitch * 0.6);
        v.vx = Math.sin(v.yaw) * horizSpeed;
        v.vz = Math.cos(v.yaw) * horizSpeed;
        v.x += v.vx * safeDt;
        v.z += v.vz * safeDt;
        v.y += v.vy * safeDt;
        v.roll = -turnInput * 0.42;

        if (v.y <= 0.1) {
          v.y = 0.1;
          v.vy = 0;
          v.grounded = true;
          v.pitch = 0;
        } else {
          v.grounded = false;
        }
      } else if (v.kind === 'helicopter') {
        if (v.rotorMain) v.rotorMain.rotation.y += 34 * safeDt;
        if (v.rotorTail) v.rotorTail.rotation.x += 42 * safeDt;

        if (climbHeld) {
          v.vy = Math.min(18, v.vy + (spaceHeld ? 28 : 22) * safeDt);
        } else if (descendHeld) {
          v.vy = Math.max(-13, v.vy - 20 * safeDt);
        } else {
          v.vy += (0 - v.vy) * 4.5 * safeDt;
        }

        const horizAccel = v.y > 0.5 ? v.accel * 0.85 : v.accel * 0.55;
        if (forwardInput !== 0) {
          v.speed = Math.max(-22, Math.min(v.maxSpeed, v.speed + forwardInput * horizAccel * safeDt));
        } else {
          v.speed *= Math.pow(0.35, safeDt);
        }

        v.yaw += turnInput * 1.85 * safeDt;
        v.vx = Math.sin(v.yaw) * v.speed;
        v.vz = Math.cos(v.yaw) * v.speed;
        v.x += v.vx * safeDt;
        v.z += v.vz * safeDt;
        v.y += v.vy * safeDt;
        v.pitch = forwardInput * 0.22;
        v.roll = -turnInput * 0.28;

        if (v.y <= 0.1) {
          v.y = 0.1;
          v.vy = 0;
          v.grounded = true;
        } else {
          v.grounded = false;
        }
      } else if (v.kind === 'boat') {
        const onWater = isDeepWaterAt(v.x, v.z);
        const maxBoatSpd = onWater ? v.maxSpeed * boostMult : 9;
        if (forwardInput !== 0) {
          v.speed = Math.max(-14, Math.min(maxBoatSpd, v.speed + forwardInput * v.accel * boostMult * safeDt));
        } else {
          v.speed *= Math.pow(0.45, safeDt);
        }
        v.yaw += turnInput * 1.5 * safeDt;
        v.vx = Math.sin(v.yaw) * v.speed;
        v.vz = Math.cos(v.yaw) * v.speed;
        v.x += v.vx * safeDt;
        v.z += v.vz * safeDt;
        v.y = onWater ? 0.15 + Math.sin(now * 0.004 + v.id) * 0.08 : 0.1;
      } else {
        if (forwardInput !== 0) {
          v.speed = Math.max(-20, Math.min(v.maxSpeed * boostMult, v.speed + forwardInput * v.accel * boostMult * safeDt));
        } else {
          v.speed *= Math.pow(0.25, safeDt);
        }
        const reverseSign = v.speed >= -0.2 ? 1 : -1;
        if (Math.abs(v.speed) > 0.4) {
          v.yaw += turnInput * (v.kind === 'tank' ? 1.15 : 2.15) * reverseSign * safeDt;
        }
        v.vx = Math.sin(v.yaw) * v.speed;
        v.vz = Math.cos(v.yaw) * v.speed;
        const nextX = Math.max(-340, Math.min(340, v.x + v.vx * safeDt));
        const nextZ = Math.max(-340, Math.min(340, v.z + v.vz * safeDt));

        if (isDeepWaterAt(nextX, nextZ)) {
          v.speed = -v.speed * 0.25;
          showToast('🚫 Samochód nie może wjechać do wody!');
        } else {
          v.x = nextX;
          v.z = nextZ;
        }
        v.y = 0.1;

        if (Math.abs(v.speed) > 10) {
          npcs.forEach((n) => {
            if (Math.hypot(n.x - v.x, n.z - v.z) < 2.5 && n.ragdollTimer <= 0) {
              triggerRagdoll(n, v.vx * 0.45, 5.5, v.vz * 0.45, 2.4);
              n.hp -= 45;
              if (n.role !== 'civilian') {
                state.wantedLevel = Math.min(6, state.wantedLevel + 1);
              }
            }
          });
        }
      }

      v.x = Math.max(-340, Math.min(340, v.x));
      v.z = Math.max(-340, Math.min(340, v.z));
      v.mesh.position.set(v.x, v.y, v.z);
      v.mesh.rotation.y = v.yaw;
      v.mesh.rotation.x = v.pitch || 0;
      v.mesh.rotation.z = v.roll || 0;

      state.player.x = v.x;
      state.player.y = v.y + 1.2;
      state.player.z = v.z;
      playerGroup.visible = false;

      $('hudVehicleBox').classList.remove('hidden');
      $('vehNameLabel').textContent = v.name;
      $('vehSpeedVal').textContent = `${Math.round(Math.abs(v.speed) * 3.4)} km/h`;
      $('vehFuelNitroVal').textContent = `${Math.round(v.fuel)}% / ${Math.round(v.nitro)}%`;
      if (v.kind === 'airplane' || v.kind === 'helicopter') {
        $('vehAeroRow').classList.remove('hidden');
        $('vehAeroVal').textContent = `${Math.max(0, v.y).toFixed(1)}m / ${Math.round((v.pitch * 180) / Math.PI)}°`;
        $('vehControlsHint').textContent = '[SPACJA / R] Wznoszenie • [Q / Shift] Opadanie • [W/S] Ciąg • [A/D] Ster • [E/F] Wyjdź';
      } else {
        $('vehAeroRow').classList.add('hidden');
        $('vehControlsHint').textContent = '[W/S] Jazda • [A/D] Skręt • [N] Nitro • [E/F] Wysiądź';
      }
    } else {
      $('hudVehicleBox').classList.add('hidden');
      playerGroup.visible = !state.firstPersonCam;

      let moveF = (keys['KeyW'] ? 1 : 0) - (keys['KeyS'] ? 1 : 0);
      let moveR = (keys['KeyD'] ? 1 : 0) - (keys['KeyA'] ? 1 : 0);
      const len = Math.hypot(moveF, moveR);
      if (len > 0) {
        moveF /= len;
        moveR /= len;
      }

      if (!state.player.grounded && !state.onMoon && state.player.vy < -6.5 && keys['Space'] && !state.parachuteOpen) {
        deployParachute();
      }

      const sprint = (Boolean(keys['ShiftLeft']) || Boolean(keys['ShiftRight'])) && state.energy > 2;
      if (sprint && len > 0) {
        state.energy = Math.max(0, state.energy - 8 * safeDt);
      } else {
        state.energy = Math.min(100, state.energy + 6 * safeDt);
      }

      const rolling = state.player.rollTimer > 0;
      if (rolling) state.player.rollTimer = Math.max(0, state.player.rollTimer - safeDt);

      const spd = state.parachuteOpen ? 6.2 : (rolling ? 16.5 : (sprint ? 14.5 : 8.8));
      const forwardX = Math.sin(state.player.camYaw);
      const forwardZ = Math.cos(state.player.camYaw);
      const rightX = Math.cos(state.player.camYaw);
      const rightZ = -Math.sin(state.player.camYaw);

      const targetVx = (forwardX * moveF + rightX * moveR) * spd;
      const targetVz = (forwardZ * moveF + rightZ * moveR) * spd;
      state.player.vx += (targetVx - state.player.vx) * 12 * safeDt;
      state.player.vz += (targetVz - state.player.vz) * 12 * safeDt;

      if (keys['Space'] && state.player.grounded) {
        state.player.vy = state.onMoon ? 11.5 : 8.4;
        state.player.grounded = false;
      }

      const gravity = state.onMoon ? 5.8 : 22.0;
      state.player.vy -= gravity * safeDt;

      if (state.parachuteOpen) {
        state.player.vy = Math.max(-4.8, state.player.vy);
      }

      if (!state.player.grounded && state.player.vy < 0) {
        state.player.maxFallSpeed = Math.max(state.player.maxFallSpeed, Math.abs(state.player.vy));
      }

      const nextPX = state.player.x + state.player.vx * safeDt;
      const nextPZ = state.player.z + state.player.vz * safeDt;

      if (!state.onMoon && !state.insideBuildingId && state.player.y <= 0.2 && isDeepWaterAt(nextPX, nextPZ)) {
        state.player.vx = 0;
        state.player.vz = 0;
        showToast('🌊 Głęboka woda w zatoce — użyj łodzi [B] lub mostu!');
      } else {
        state.player.x = nextPX;
        state.player.z = nextPZ;
      }

      state.player.y += state.player.vy * safeDt;

      if (state.insideBuildingId) {
        const b = buildings.find((item) => item.id === state.insideBuildingId);
        if (b && b.interiorBounds) {
          state.player.x = Math.max(b.interiorBounds.minX + 1.2, Math.min(b.interiorBounds.maxX - 1.2, state.player.x));
          state.player.z = Math.max(b.interiorBounds.minZ + 1.2, Math.min(b.interiorBounds.maxZ - 1.2, state.player.z));
        }
      } else {
        state.player.x = Math.max(-340, Math.min(340, state.player.x));
        state.player.z = Math.max(-340, Math.min(340, state.player.z));
      }

      if (state.player.y <= 0) {
        if (!state.player.grounded && !state.onMoon) {
          if (state.parachuteOpen) {
            addLog('Bezpieczne lądowanie na spadochronie!', 'text-sky-300');
          } else if (state.player.maxFallSpeed > 15.5) {
            const fallDmg = Math.min(85, Math.round((state.player.maxFallSpeed - 15.5) * 4.2));
            let remDmg = fallDmg;
            if (state.armor > 0) {
              const absorbed = Math.min(state.armor, remDmg);
              state.armor -= absorbed;
              remDmg -= absorbed;
            }
            state.hp = Math.max(5, state.hp - remDmg);
            SoundFX.tone(95, 0.22, 'sawtooth', 0.09);
            addLog(`Obrażenia od upadku: -${fallDmg} (użyj [O/Spacji] dla spadochronu)!`, 'text-rose-400');
            showToast(`💥 Upadek: -${fallDmg} obrażeń!`);
          }
        }
        state.player.y = 0;
        state.player.vy = 0;
        state.player.grounded = true;
        state.player.maxFallSpeed = 0;
        state.parachuteOpen = false;
        if (parachuteMesh) parachuteMesh.visible = false;
      }

      if (parachuteMesh) parachuteMesh.visible = state.parachuteOpen;
      if ($('hudParachuteBadge')) $('hudParachuteBadge').classList.toggle('hidden', !state.parachuteOpen);

      if (Math.hypot(state.player.vx, state.player.vz) > 0.6) {
        state.player.yaw = Math.atan2(state.player.vx, state.player.vz);
        playerLimbs.phase += safeDt * 12;
      } else {
        playerLimbs.phase = 0;
      }

      playerGroup.position.set(state.player.x, state.player.y, state.player.z);
      playerGroup.rotation.y = state.player.yaw;
      playerGroup.rotation.x = rolling ? ((1 - state.player.rollTimer / 0.45) * Math.PI * 2) : 0;
      playerLimbs.leftLeg.rotation.x = Math.sin(playerLimbs.phase) * 0.65;
      playerLimbs.rightLeg.rotation.x = -Math.sin(playerLimbs.phase) * 0.65;
    }

    for (let i = 0; i < vehicles.length; i++) {
      const v = vehicles[i];
      if (!v.aiControlled || v.id === state.inVehicleId) continue;

      v.aiTimer -= safeDt;
      if (v.aiTimer <= 0) {
        v.aiTimer = 2.0 + Math.random() * 2.5;
        v.yaw += (Math.random() - 0.5) * 1.1;
      }

      if (v.kind === 'airplane') {
        if (v.propeller) v.propeller.rotation.z += 28 * safeDt;
        v.speed = 34;
        v.y = 32 + Math.sin(now * 0.0012 + v.id) * 6;
        v.vx = Math.sin(v.yaw) * v.speed;
        v.vz = Math.cos(v.yaw) * v.speed;
        v.x += v.vx * safeDt;
        v.z += v.vz * safeDt;
        if (Math.hypot(v.x, v.z) > 260) v.yaw = Math.atan2(-v.x, -v.z);
        v.mesh.position.set(v.x, v.y, v.z);
        v.mesh.rotation.y = v.yaw;
        v.mesh.rotation.z = -0.14;
      } else if (v.kind === 'helicopter') {
        if (v.rotorMain) v.rotorMain.rotation.y += 32 * safeDt;
        if (v.rotorTail) v.rotorTail.rotation.x += 40 * safeDt;
        v.speed = 19;
        v.y = 22 + Math.cos(now * 0.0014 + v.id) * 5;
        v.vx = Math.sin(v.yaw) * v.speed;
        v.vz = Math.cos(v.yaw) * v.speed;
        v.x += v.vx * safeDt;
        v.z += v.vz * safeDt;
        if (Math.hypot(v.x, v.z) > 240) v.yaw = Math.atan2(-v.x, -v.z);
        v.mesh.position.set(v.x, v.y, v.z);
        v.mesh.rotation.y = v.yaw;
      } else if (v.kind === 'tank') {
        v.speed = 11;
        const nx = v.x + Math.sin(v.yaw) * v.speed * safeDt;
        const nz = v.z + Math.cos(v.yaw) * v.speed * safeDt;
        if (isDeepWaterAt(nx, nz) || Math.abs(nx) > 190 || Math.abs(nz) > 190) {
          v.yaw += Math.PI * 0.65;
        } else {
          v.x = nx;
          v.z = nz;
        }
        v.y = 0.1;
        v.mesh.position.set(v.x, v.y, v.z);
        v.mesh.rotation.y = v.yaw;
      } else if (v.kind === 'car') {
        v.speed = 17;
        const nx = v.x + Math.sin(v.yaw) * v.speed * safeDt;
        const nz = v.z + Math.cos(v.yaw) * v.speed * safeDt;
        if (isDeepWaterAt(nx, nz) || Math.abs(nx) > 210 || Math.abs(nz) > 210) {
          v.yaw += Math.PI * 0.7;
        } else {
          v.x = nx;
          v.z = nz;
        }
        v.y = 0.1;
        v.mesh.position.set(v.x, v.y, v.z);
        v.mesh.rotation.y = v.yaw;
      }
    }

    for (let i = 0; i < npcs.length; i++) {
      const n = npcs[i];
      if (!n || !n.mesh) continue;

      if (n.ragdollTimer > 0) {
        n.ragdollTimer -= safeDt;
        n.vy -= 20.0 * safeDt;
        n.x += n.vx * safeDt;
        n.z += n.vz * safeDt;
        n.y += n.vy * safeDt;
        n.vx *= Math.pow(0.35, safeDt);
        n.vz *= Math.pow(0.35, safeDt);

        if (n.y <= 0) {
          n.y = 0;
          n.vy = Math.abs(n.vy) > 2.2 ? -n.vy * 0.35 : 0;
        }
        n.mesh.position.set(n.x, n.y, n.z);
        n.mesh.rotation.x = Math.min(Math.PI / 2, n.mesh.rotation.x + 6.5 * safeDt);
        n.leftLeg.rotation.x = 0.6;
        n.rightLeg.rotation.x = -0.5;
        if (n.ragdollTimer <= 0) {
          n.mesh.rotation.x = 0;
          n.leftLeg.rotation.x = 0;
          n.rightLeg.rotation.x = 0;
        }
        continue;
      }

      n.aiTurnTimer -= safeDt;
      if (n.aiTurnTimer <= 0) {
        n.aiTurnTimer = 2.0 + Math.random() * 3.0;
        n.yaw = Math.random() * Math.PI * 2;
      }
      const walkSpd = n.role === 'civilian' ? 3.2 : 5.2;
      const nx = n.x + Math.sin(n.yaw) * walkSpd * safeDt;
      const nz = n.z + Math.cos(n.yaw) * walkSpd * safeDt;
      if (!isDeepWaterAt(nx, nz)) {
        n.x = Math.max(-230, Math.min(230, nx));
        n.z = Math.max(-230, Math.min(230, nz));
      } else {
        n.yaw += Math.PI * 0.5;
      }
      n.walkPhase += safeDt * 9;
      n.mesh.position.set(n.x, 0, n.z);
      n.mesh.rotation.y = n.yaw;
      n.leftLeg.rotation.x = Math.sin(n.walkPhase) * 0.5;
      n.rightLeg.rotation.x = -Math.sin(n.walkPhase) * 0.5;
    }

    for (let i = projectiles.length - 1; i >= 0; i--) {
      const p = projectiles[i];
      p.life -= safeDt;
      p.mesh.position.x += p.vx * safeDt;
      p.mesh.position.y += p.vy * safeDt;
      p.mesh.position.z += p.vz * safeDt;

      let hit = false;
      for (let j = 0; j < npcs.length; j++) {
        const n = npcs[j];
        if (Math.hypot(n.x - p.mesh.position.x, n.z - p.mesh.position.z) < 1.35 && Math.abs(p.mesh.position.y - 1.3) < 1.5) {
          hit = true;
          n.hp -= p.dmg;
          triggerRagdoll(n, p.vx * 0.14, 5.2, p.vz * 0.14, 2.3);
          if (n.role !== 'civilian' || state.wantedLevel === 0) {
            state.wantedLevel = Math.min(6, state.wantedLevel + 1);
            state.wantedDecayTimer = 0;
          }
          if (n.hp <= 0) {
            state.cash += 100;
            n.hp = 100;
            n.x = (Math.random() - 0.5) * 180;
            n.z = (Math.random() - 0.5) * 180;
          }
          break;
        }
      }

      if (hit || p.life <= 0 || p.mesh.position.y <= 0.1) {
        if (p.explosive) {
          spawnExplosion(p.mesh.position.x, p.mesh.position.y, p.mesh.position.z, 9.5);
        }
        disposeObject3D(p.mesh);
        projectiles.splice(i, 1);
      }
    }

    for (let i = particles.length - 1; i >= 0; i--) {
      const pt = particles[i];
      pt.life -= safeDt;
      pt.vy -= 16 * safeDt;
      pt.mesh.position.x += pt.vx * safeDt;
      pt.mesh.position.y = Math.max(0.2, pt.mesh.position.y + pt.vy * safeDt);
      pt.mesh.position.z += pt.vz * safeDt;
      if (pt.life <= 0) {
        disposeObject3D(pt.mesh);
        particles.splice(i, 1);
      }
    }

    for (let i = supplyDrops.length - 1; i >= 0; i--) {
      const s = supplyDrops[i];
      if (s.y > 0.8) {
        s.y = Math.max(0.8, s.y + s.vy * safeDt);
        s.mesh.position.y = s.y;
      }
      if (Math.hypot(s.x - state.player.x, s.z - state.player.z) < 3.2) {
        state.cash += 400;
        state.hp = Math.min(state.maxHp, state.hp + 35);
        state.armor = Math.min(100, state.armor + 40);
        SoundFX.pickup();
        addLog('Odebrano zrzut zaopatrzenia: +$400, +35 HP, +40 Kamizelka!', 'text-emerald-300');
        disposeObject3D(s.mesh);
        supplyDrops.splice(i, 1);
      }
    }

    if (state.firstPersonCam && !activeVeh) {
      camera.position.set(state.player.x, state.player.y + 1.85, state.player.z);
      const lookX = state.player.x + Math.sin(state.player.camYaw) * 12;
      const lookY = state.player.y + 1.75 - Math.sin(state.player.camPitch) * 5;
      const lookZ = state.player.z + Math.cos(state.player.camYaw) * 12;
      camera.lookAt(lookX, lookY, lookZ);
    } else {
      const camDist = activeVeh ? (activeVeh.kind === 'airplane' || activeVeh.kind === 'helicopter' ? 14.5 : 8.5) : 6.8;
      const camHeight = activeVeh ? (activeVeh.kind === 'airplane' || activeVeh.kind === 'helicopter' ? 4.4 : 2.8) : 2.5;
      const targetCamX = state.player.x - Math.sin(state.player.camYaw) * camDist;
      const targetCamZ = state.player.z - Math.cos(state.player.camYaw) * camDist;
      const targetCamY = Math.max(1.4, state.player.y + camHeight + Math.sin(state.player.camPitch) * 4.2);

      camera.position.lerp(new THREE.Vector3(targetCamX, targetCamY, targetCamZ), 1 - Math.exp(-10 * safeDt));
      camera.lookAt(state.player.x, state.player.y + 1.5, state.player.z);
    }

    const ctxPrompt = getContextualPrompt();
    const ctxBar = $('hudContextBar');
    if (ctxBar) {
      if (ctxPrompt) {
        ctxBar.textContent = ctxPrompt;
        ctxBar.classList.remove('hidden');
      } else {
        ctxBar.classList.add('hidden');
      }
    }

    updateHudLabels();
    BuiltinOptimizer.update(safeDt, { x: state.player.x, y: state.player.y, z: state.player.z });
  }

    $('hudRadioLabel').textContent = `🎵 Radio: ${SoundFX.radioStations[SoundFX.radioStationIdx]} [M]`;
  }

          <div class="font-bold text-white">${fItem.name}</div>

        </div>
        <button type="button" class="px-3 py-1.5 rounded-xl bg-amber-600 hover:bg-amber-500 font-bold text-white shrink-0 cursor-pointer">$100</button>
      `;
      row.querySelector('button').addEventListener('click', () => {
        if (state.cash < 100) {
          showToast('Za mało gotówki (potrzeba $100)');

        state.cash -= 100;
        if (fItem.paraAdd) state.parachutes += fItem.paraAdd;
        if (fItem.healAdd) state.hp = Math.min(state.maxHp, state.hp + fItem.healAdd);
        if (fItem.energyAdd) state.energy = Math.min(100, state.energy + fItem.energyAdd);
        SoundFX.pickup();
        updateHudLabels();
        renderInventoryModal();
        addLog(`Użyto/kupiono "${fItem.name}" za $100`, 'text-amber-300');
      });
      foodBox.appendChild(row);
    });

    shopBox.innerHTML = '';
    SHOP_ITEMS.forEach((item) => {
      const row = document.createElement('div');
      row.className = 'p-2.5 rounded-xl bg-slate-900 border border-slate-800 flex items-center justify-between gap-2';
      row.innerHTML = `
        <div>
          <div class="font-bold text-white">${item.name}</div>
          <div class="text-[11px] text-slate-400">${item.desc}</div>
        </div>
        <button type="button" class="px-3 py-1.5 rounded-xl bg-emerald-600 hover:bg-emerald-500 font-bold text-white shrink-0 cursor-pointer">Kup $100</button>
      `;
      row.querySelector('button').addEventListener('click', () => {
        if (state.cash < 100) {
          showToast('Za mało gotówki (potrzeba $100)');
          return;
        }
        state.cash -= 100;
        if (item.slot && state.weapons[item.slot]) {
          state.weapons[item.slot].ammo += item.ammoAdd || 30;
          state.activeSlot = item.slot;
        }
        if (item.armorAdd) state.armor = Math.min(100, state.armor + item.armorAdd);
        if (item.healAdd) state.hp = Math.min(state.maxHp, state.hp + item.healAdd);
        SoundFX.pickup();
        updateHudLabels();
        renderInventoryModal();
        addLog(`Kupiono "${item.name}" za $100`, 'text-emerald-300');
      });
      shopBox.appendChild(row);
    });
  }

  function renderGarageModal() {
    const q = ($('garageSearchInput')?.value || '').trim().toLowerCase();
    const cls = $('garageClassFilter')?.value || 'ALL';
    const filtered = VEHICLES_400.filter((v) => {
      const matchCls = cls === 'ALL' || v.cls === cls;
      const matchQ = !q || v.name.toLowerCase().includes(q) || v.cls.toLowerCase().includes(q) || String(v.id) === q;
      return matchCls && matchQ;
    });

    const perPage = 12;
    const totalPages = Math.max(1, Math.ceil(filtered.length / perPage));
    state.garagePage = Math.max(1, Math.min(totalPages, state.garagePage));
    const startIdx = (state.garagePage - 1) * perPage;
    const slice = filtered.slice(startIdx, startIdx + perPage);

    const grid = $('garageGrid');
    grid.innerHTML = '';
    slice.forEach((sp) => {
      const card = document.createElement('div');
      card.className = 'p-2.5 rounded-xl bg-slate-900 border border-slate-800 flex items-center justify-between gap-2';
      card.innerHTML = `
        <div class="min-w-0">
          <div class="font-bold text-white truncate">#${sp.id} ${sp.name}</div>
          <div class="text-[10px] text-cyan-300">${sp.cls} • V-max ${sp.maxSpeed} • <span class="text-emerald-400 font-bold">$0</span></div>
        </div>
        <button type="button" class="px-2.5 py-1.5 rounded-xl bg-cyan-600 hover:bg-cyan-500 font-bold text-white shrink-0 cursor-pointer">Przywołaj ($0)</button>
      `;
      card.querySelector('button').addEventListener('click', () => {
        if (state.onMoon) returnFromMoonToEarth();
        if (state.insideBuildingId) toggleBuildingInterior();
        let spawnX = state.player.x + Math.sin(state.player.yaw) * 5;
        let spawnZ = state.player.z + Math.cos(state.player.yaw) * 5;
        let spawnY = 0.1;
        if (sp.kind === 'boat') {
          spawnX = -175;
          spawnZ = 158;
          spawnY = 0.15;
        } else if (sp.kind === 'airplane') {
          spawnX = 135;
          spawnZ = -85;
          spawnY = 0.15;
        } else if (isDeepWaterAt(spawnX, spawnZ)) {
          spawnX = 0;
          spawnZ = 22;
        }
        const veh = createVehicleEntity(sp, spawnX, spawnY, spawnZ, state.player.yaw, false);
        state.inVehicleId = veh.id;
        $('garageModal').classList.add('hidden');
        addLog(`Przywołano za $0: ${sp.name}`, 'text-cyan-300');
        showToast(`Przywołano ${sp.name} za $0!`);
        updateHudLabels();
      });
      grid.appendChild(card);
    });

    $('garagePageInfo').textContent = `Strona ${state.garagePage} / ${totalPages} (${filtered.length} pojazdów)`;
  }

  function renderEncyclopediaModal() {
    const q = ($('encSearchInput')?.value || '').trim().toLowerCase();
    const cat = $('encCategorySelect')?.value || 'ALL';
    const perPage = 18;

    if (!q && cat === 'ALL') {
      const totalPages = Math.ceil(TOTAL_ENCYCLOPEDIA_COUNT / perPage);
      state.encPage = Math.max(1, Math.min(totalPages, state.encPage));
      const startId = (state.encPage - 1) * perPage + 1;
      const endId = Math.min(TOTAL_ENCYCLOPEDIA_COUNT, startId + perPage - 1);

      const list = $('encListContainer');
      list.innerHTML = '';
      for (let id = startId; id <= endId; id++) {
        const item = getEncyclopediaEntryByIndex(id);
        const div = document.createElement('div');
        div.className = 'p-2 rounded-xl bg-slate-950/90 border border-slate-800/90';
        div.innerHTML = `<div class="font-bold text-pink-300">${item.title}</div><div class="text-[11px] text-slate-400">${item.desc}</div>`;
        list.appendChild(div);
      }
      $('encFilteredCount').textContent = `${TOTAL_ENCYCLOPEDIA_COUNT.toLocaleString('pl-PL')} wyników`;
      $('encPageLabel').textContent = `Strona ${state.encPage.toLocaleString('pl-PL')} / ${totalPages.toLocaleString('pl-PL')}`;
      return;
    }

    const band = ENC_RANGES.find((r) => r.cat === cat) || { start: 1, end: TOTAL_ENCYCLOPEDIA_COUNT };
    const matches = [];
    const step = Math.max(1, Math.floor((band.end - band.start + 1) / 400));
    for (let id = band.start; id <= band.end && matches.length < 180; id += step) {
      const entry = getEncyclopediaEntryByIndex(id);
      if (!q || entry.title.toLowerCase().includes(q) || entry.desc.toLowerCase().includes(q) || String(entry.id).includes(q)) {
        matches.push(entry);
      }
    }

    const totalPages = Math.max(1, Math.ceil(matches.length / perPage));
    state.encPage = Math.max(1, Math.min(totalPages, state.encPage));
    const slice = matches.slice((state.encPage - 1) * perPage, state.encPage * perPage);

    const list = $('encListContainer');
    list.innerHTML = '';
    slice.forEach((item) => {
      const div = document.createElement('div');
      div.className = 'p-2 rounded-xl bg-slate-950/90 border border-slate-800/90';
      div.innerHTML = `<div class="font-bold text-pink-300">${item.title}</div><div class="text-[11px] text-slate-400">${item.desc}</div>`;
      list.appendChild(div);
    });

    const approxTotal = cat === 'ALL' && !q
      ? TOTAL_ENCYCLOPEDIA_COUNT
      : (cat !== 'ALL' ? (band.end - band.start + 1) : matches.length);
    $('encFilteredCount').textContent = `${approxTotal.toLocaleString('pl-PL')} wyników`;
    $('encPageLabel').textContent = `Strona ${state.encPage} / ${totalPages}`;
  }
