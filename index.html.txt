class AudioEngine {
  constructor() {
    this.ctx = null;
    this.enabled = true;
    this.radioIndex = 0;
    this.stations = [
      { name: 'Sus FM 3D — Synthwave', tempo: 118, notes: [220, 261.63, 329.63, 392, 329.63, 261.63] },
      { name: 'Los Sus Rap — Hip-Hop', tempo: 92, notes: [110, 110, 130.81, 146.83, 110, 98] },
      { name: 'WCTR 3D News Radio', tempo: 60, notes: [] },
      { name: 'Radio Wyłączone', tempo: 0, notes: [] }
    ];
    this.beatTimer = 0;
    this.noteStep = 0;
  }

  init() {
    if (!this.ctx) {
      const AudioCtx = window.AudioContext || window.webkitAudioContext;
      if (AudioCtx) this.ctx = new AudioCtx();
    }
    if (this.ctx && this.ctx.state === 'suspended') this.ctx.resume();
  }

  playTone(freq, type = 'sawtooth', duration = 0.12, vol = 0.07, slideTo = null) {
    if (!this.enabled || !this.ctx) return;
    try {
      const osc = this.ctx.createOscillator();
      const gain = this.ctx.createGain();
      osc.type = type;
      osc.frequency.setValueAtTime(freq, this.ctx.currentTime);
      if (slideTo) osc.frequency.exponentialRampToValueAtTime(slideTo, this.ctx.currentTime + duration);
      gain.gain.setValueAtTime(vol, this.ctx.currentTime);
      gain.gain.exponentialRampToValueAtTime(0.001, this.ctx.currentTime + duration);
      osc.connect(gain);
      gain.connect(this.ctx.destination);
      osc.start();
      osc.stop(this.ctx.currentTime + duration);
    } catch (e) {}
  }

  playNoise(duration = 0.2, vol = 0.12) {
    if (!this.enabled || !this.ctx) return;
    try {
      const bufferSize = Math.floor(this.ctx.sampleRate * duration);
      const buffer = this.ctx.createBuffer(1, bufferSize, this.ctx.sampleRate);
      const data = buffer.getChannelData(0);
      for (let i = 0; i < bufferSize; i++) data[i] = Math.random() * 2 - 1;
      const noise = this.ctx.createBufferSource();
      noise.buffer = buffer;
      const gain = this.ctx.createGain();
      gain.gain.setValueAtTime(vol, this.ctx.currentTime);
      gain.gain.exponentialRampToValueAtTime(0.001, this.ctx.currentTime + duration);
      noise.connect(gain);
      gain.connect(this.ctx.destination);
      noise.start();
    } catch (e) {}
  }

  shoot(weaponType) {
    this.init();
    if (weaponType === 'melee') this.playTone(170, 'triangle', 0.09, 0.08, 65);
    else if (weaponType === 'silenced') this.playTone(430, 'sine', 0.06, 0.05, 110);
    else if (weaponType === 'rpg' || weaponType === 'tank') {
      this.playNoise(0.45, 0.22);
      this.playTone(115, 'sawtooth', 0.4, 0.16, 32);
    } else if (weaponType === 'flame') {
      this.playNoise(0.14, 0.08);
    } else {
      this.playNoise(0.1, 0.13);
      this.playTone(250, 'square', 0.08, 0.08, 60);
    }
  }

  explosion() {
    this.init();
    this.playNoise(0.65, 0.26);
    this.playTone(90, 'sawtooth', 0.55, 0.2, 24);
  }

  cash() {
    this.init();
    this.playTone(880, 'sine', 0.09, 0.09);
    setTimeout(() => this.playTone(1320, 'sine', 0.14, 0.09), 75);
  }

  sirenTick(time) {
    if (!this.enabled || !this.ctx) return;
    const freq = Math.sin(time * 7) > 0 ? 780 : 590;
    this.playTone(freq, 'triangle', 0.14, 0.02);
  }

  updateRadio(dt, inVehicle) {
    if (!inVehicle || !this.enabled || !this.ctx) return;
    const st = this.stations[this.radioIndex];
    if (!st || st.notes.length === 0) return;
    this.beatTimer += dt;
    const interval = 60 / st.tempo;
    if (this.beatTimer >= interval) {
      this.beatTimer = 0;
      const note = st.notes[this.noteStep % st.notes.length];
      this.playTone(note, 'triangle', interval * 0.8, 0.02);
      this.noteStep++;
    }
  }
}

async function askGroqNpcReply(npc, promptText) {
  const apiKey = getNextGroqApiKey();
  const controller = new AbortController();
  const timeout = setTimeout(() => controller.abort(), 3200);
  try {
    const res = await fetch('https://api.groq.com/openai/v1/chat/completions', {
      method: 'POST',
      signal: controller.signal,
      headers: {
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${apiKey}`
      },
      body: JSON.stringify({
        model: 'llama-3.1-8b-instant',
        max_tokens: 90,
        temperature: 0.75,
        messages: [
          {
            role: 'system',
            content: `Jesteś postacią NPC o imieniu ${npc.name} (rola: ${npc.role}) w grze GTA SUS 3D. Odpowiadaj krótko, po polsku, w klimacie otwartego miasta (1-2 zdania).`
          },
          { role: 'user', content: promptText }
        ]
      })
    });
    clearTimeout(timeout);
    if (!res.ok) throw new Error('HTTP ' + res.status);
    const data = await res.json();
    const text = data?.choices?.[0]?.message?.content?.trim();
    if (!text) throw new Error('Empty reply');
    return { text, source: `Groq API #${((groqKeyCursor - 1) % GROQ_API_KEYS.length) + 1}` };
  } catch (e) {
    clearTimeout(timeout);
    return { text: getOfflineNpcFallback(npc, promptText), source: 'Tryb Offline AI' };
  }
}

  for (let i = 0; i < 400; i++) {
    const base = BASE_VEHICLE_TEMPLATES[i % BASE_VEHICLE_TEMPLATES.length];
    const code = String(i + 1).padStart(3, '0');
    const col = i < BASE_VEHICLE_TEMPLATES.length ? base.color : palette[i % palette.length];
    const name = i < BASE_VEHICLE_TEMPLATES.length
      ? `${base.name} #${code}`
      : `${base.name.split(' (')[0]} ${suffixes[i % suffixes.length]} #${code}`;

    arr.push({
      ...base,
      id: `${base.id}_${i}`,
      baseId: base.id,
      code,
      name,
      color: col,
      price: 0,
      maxSpeed: Math.round(base.maxSpeed * (0.9 + ((i * 7) % 25) / 100))
    });
  }
  return arr;
}

function applyGraphicsPreset() {
  if (!renderer || !dirLight) return;
  const preset = Settings.graphicsPreset;
  let capDpr = 1.5;
  let shadowRes = 2048;
  if (preset === 'low') { capDpr = 1; shadowRes = 512; }
  else if (preset === 'medium') { capDpr = 1.25; shadowRes = 1024; }
  else if (preset === 'high') { capDpr = 1.75; shadowRes = 2048; }
  else if (preset === 'ultra') { capDpr = 2.0; shadowRes = 4096; }

  renderer.setPixelRatio(Math.min(window.devicePixelRatio, capDpr));
  renderer.toneMapping = THREE.ACESFilmicToneMapping;
  renderer.toneMappingExposure = Settings.exposure;
  renderer.shadowMap.enabled = Settings.shadowsEnabled && preset !== 'low';
  renderer.shadowMap.type = THREE.PCFSoftShadowMap;
  dirLight.castShadow = renderer.shadowMap.enabled;
  if (dirLight.shadow.mapSize.width !== shadowRes) {
    dirLight.shadow.mapSize.set(shadowRes, shadowRes);
    if (dirLight.shadow.map) {
      dirLight.shadow.map.dispose();
      dirLight.shadow.map = null;
    }
  }
}

function buildClouds3D() {
  clouds3D.forEach(c => scene.remove(c.mesh));
  clouds3D = [];
  const cloudMat = new THREE.MeshStandardMaterial({ color: 0xffffff, roughness: 0.95, transparent: true, opacity: 0.88 });
  const maxClouds = 44;
  for (let i = 0; i < maxClouds; i++) {
    const group = new THREE.Group();
    const p1 = new THREE.Mesh(new THREE.BoxGeometry(16, 4.5, 10), cloudMat);
    const p2 = new THREE.Mesh(new THREE.BoxGeometry(10, 6.5, 8), cloudMat);
    p2.position.set(2, 1.8, 0);
    const p3 = new THREE.Mesh(new THREE.BoxGeometry(9, 4.0, 7), cloudMat);
    p3.position.set(-6, -0.4, 1);
    group.add(p1, p2, p3);
    const layerY = i % 2 === 0 ? 74 + (i % 4) * 9 : 108 + (i % 3) * 11;
    group.position.set(
      (Math.random() - 0.5) * 780,
      layerY,
      (Math.random() - 0.5) * 780
    );
    scene.add(group);
    clouds3D.push({ mesh: group, speed: 3 + (i % 6) * 1.3 });
  }
  syncCloudDensity();
}

function syncCloudDensity() {
  const activeCount = Math.round((Settings.cloudDensity / 100) * clouds3D.length);
  clouds3D.forEach((c, idx) => {
    c.mesh.visible = idx < activeCount;
  });
}

function spawnVehicle3D(typeOrId, x, z, yaw = 0, options = {}) {
  const spec = VEHICLES_400.find(v => v.id === typeOrId || v.baseId === typeOrId) || VEHICLES_400[3];
  const group = new THREE.Group();

  const bodyMat = new THREE.MeshStandardMaterial({ color: spec.color, metalness: 0.35, roughness: 0.35 });
  const body = new THREE.Mesh(new THREE.BoxGeometry(spec.w, spec.h * 0.65, spec.l), bodyMat);
  body.position.y = spec.bigWheels ? 1.5 : spec.airplane ? 1.2 : 0.75;
  body.castShadow = true;
  body.receiveShadow = true;
  group.add(body);

  if (spec.wheels === 4 && !spec.tank) {
    const cabin = new THREE.Mesh(
      new THREE.BoxGeometry(spec.w * 0.82, spec.h * 0.55, spec.l * 0.52),
      new THREE.MeshStandardMaterial({ color: 0x0f172a, metalness: 0.7, roughness: 0.2 })
    );
    cabin.position.set(0, (spec.bigWheels ? 1.5 : 0.75) + spec.h * 0.5, -spec.l * 0.05);
    cabin.castShadow = true;
    group.add(cabin);
  }

  if (spec.airplane) {
    const wings = new THREE.Mesh(new THREE.BoxGeometry(15.5, 0.25, 3.2), new THREE.MeshStandardMaterial({ color: 0x38bdf8 }));
    wings.position.set(0, 1.35, 0);
    wings.castShadow = true;
    group.add(wings);

    const tail = new THREE.Mesh(new THREE.BoxGeometry(5.2, 2.2, 1.4), bodyMat);
    tail.position.set(0, 2.1, -spec.l * 0.42);
    group.add(tail);
  }

  if (spec.waterOnly) {
    const bow = new THREE.Mesh(new THREE.ConeGeometry(spec.w * 0.55, 2.4, 4), bodyMat);
    bow.rotation.x = Math.PI / 2;
    bow.rotation.y = Math.PI / 4;
    bow.position.set(0, 0.75, spec.l * 0.5 + 0.9);
    group.add(bow);
  }

  const wheelMat = new THREE.MeshStandardMaterial({ color: 0x090d16, roughness: 0.8 });
  const wRadius = spec.bigWheels ? 0.95 : spec.wheels === 2 ? 0.42 : 0.48;
  if (spec.wheels === 4) {
    const offsets = [
      [-spec.w * 0.52, wRadius, spec.l * 0.32],
      [spec.w * 0.52, wRadius, spec.l * 0.32],
      [-spec.w * 0.52, wRadius, -spec.l * 0.32],
      [spec.w * 0.52, wRadius, -spec.l * 0.32]
    ];
    offsets.forEach(([wx, wy, wz]) => {
      const wheel = new THREE.Mesh(new THREE.CylinderGeometry(wRadius, wRadius, 0.38, 12), wheelMat);
      wheel.rotation.z = Math.PI / 2;
      wheel.position.set(wx, wy, wz);
      group.add(wheel);
    });
  } else if (spec.wheels === 2) {
    [spec.l * 0.35, -spec.l * 0.35].forEach(wz => {
      const wheel = new THREE.Mesh(new THREE.CylinderGeometry(wRadius, wRadius, 0.28, 12), wheelMat);
      wheel.rotation.z = Math.PI / 2;
      wheel.position.set(0, wRadius, wz);
      group.add(wheel);
    });
  }

  if (spec.tank) {
    const turret = new THREE.Mesh(new THREE.BoxGeometry(2.1, 0.9, 2.6), bodyMat);
    turret.position.set(0, 1.75, 0);
    group.add(turret);
    const barrel = new THREE.Mesh(new THREE.CylinderGeometry(0.18, 0.22, 4.2, 8), bodyMat);
    barrel.rotation.x = Math.PI / 2;
    barrel.position.set(0, 1.8, 2.8);
    group.add(barrel);
  }

  let rotorMesh = null;
  if (spec.heli) {
    rotorMesh = new THREE.Mesh(new THREE.BoxGeometry(8.5, 0.12, 0.45), new THREE.MeshStandardMaterial({ color: 0xe2e8f0 }));
    rotorMesh.position.set(0, 2.3, 0);
    group.add(rotorMesh);
  }

  let sirenMesh = null;
  if (spec.siren) {
    sirenMesh = new THREE.Mesh(
      new THREE.BoxGeometry(spec.w * 0.65, 0.26, 0.35),
      new THREE.MeshBasicMaterial({ color: 0xef4444 })
    );
    sirenMesh.position.set(0, spec.h + 0.55, 0);
    group.add(sirenMesh);
  }

  const headLight = new THREE.Mesh(new THREE.BoxGeometry(spec.w * 0.7, 0.22, 0.15), new THREE.MeshBasicMaterial({ color: 0xfef08a }));
  headLight.position.set(0, 0.75, spec.l * 0.5 + 0.05);
  group.add(headLight);

  group.position.set(x, 0, z);
  group.rotation.y = yaw;
  scene.add(group);

  const plates = ['SUS3D', 'GTASUS', 'FREE0', 'PATROL', 'AI3D', 'VAULT'];
  const vehObj = {
    ...spec,
    uid: Math.random().toString(36).slice(2),
    x, y: 0, z,
    yaw,
    pitch: 0,
    speed: 0,
    hp: spec.hp,
    maxHp: spec.hp,
    fuel: 100,
    plate: `${plates[Math.floor(Math.random() * plates.length)]}-${Math.floor(10 + Math.random() * 89)}`,
    sirenOn: false,
    driver: null,
    aiDriver: options.aiDriver || null,

function addWantedStars(amount, reason = '') {
  if (Settings.witnessSystem && GameState.wantedLevel === 0 && amount === 1) {
    const witness = npcs3D.find(n => n.role === 'civilian' && n.hp > 0 && Math.hypot(n.x - Player.x, n.z - Player.z) < 48);
    if (!witness) {
      logEvent('🤫 Brak świadków w pobliżu — nikt nie wezwał policji!', 'text-emerald-400');
      return;
    }
    if (!GameState.witnessCallingNpc) {
      GameState.witnessCallingNpc = witness;
      GameState.witnessCallTimer = 4.0;
      witness.state = 'call911';
      logEvent(`📱 Świadek (${witness.name}) widział przestępstwo (${reason}) i dzwoni na 911!`, 'text-amber-400');
      return;
    }
  }

  const prev = GameState.wantedLevel;
  GameState.wantedLevel = Math.min(6, Math.max(GameState.wantedLevel, prev + amount));
  GameState.wantedDecayTimer = 22;

  if (GameState.wantedLevel > prev) {
    const agency = GameState.wantedLevel >= 6 ? 'WOJSKO (CZOŁGI RHINO 3D)' :
                   GameState.wantedLevel >= 4 ? 'AGENCI FBI (GRANGER SUV)' :
                   GameState.wantedLevel === 3 ? 'ODDZIAŁY SWAT' : 'POLICJA LSPD';
    logEvent(`🚨 POŚCIG ${GameState.wantedLevel}★ — Wkracza ${agency}!`, 'text-red-400 font-bold');
    GameState.newsHeadline = `PILNE 3D: Pościg ${GameState.wantedLevel}★! ${agency} w akcji w ${getCurrentCityName()}!`;
    spawnPursuitUnits3D();
  }
}

function spawnPursuitUnits3D() {
  const wl = GameState.wantedLevel;
  if (wl <= 0) return;
  const ang = Math.random() * Math.PI * 2;
  const dist = 75;
  const sx = Math.max(-WORLD_HALF + 20, Math.min(WORLD_HALF - 20, Player.x + Math.sin(ang) * dist));
  const sz = Math.max(-WORLD_HALF + 20, Math.min(WORLD_HALF - 20, Player.z + Math.cos(ang) * dist));

  if (wl <= 2) {
    const copCar = spawnVehicle3D('police', sx, sz, ang + Math.PI);
    copCar.sirenOn = true;
    spawnNpc3D('police', sx + 3, sz);
    spawnNpc3D('police', sx - 3, sz);
  } else if (wl === 3) {
    const swatVan = spawnVehicle3D('swat', sx, sz, ang + Math.PI);
    swatVan.sirenOn = true;
    spawnNpc3D('swat', sx + 3, sz);
    spawnNpc3D('swat', sx - 3, sz);
  } else if (wl === 4 || wl === 5) {
    const fbiSuv = spawnVehicle3D('fbi', sx, sz, ang + Math.PI);
    fbiSuv.sirenOn = true;
    spawnNpc3D('fbi', sx + 3, sz);
    spawnNpc3D('fbi', sx - 3, sz);
    if (wl === 5) spawnNpc3D('fbi', sx, sz + 4);
  } else if (wl >= 6) {
    spawnVehicle3D('tank', sx, sz, ang + Math.PI);
    spawnNpc3D('army', sx + 4, sz);
    spawnNpc3D('fbi', sx - 4, sz);
  }
}

function triggerBankHeistAction() {
  const distVault = Math.hypot(Player.x - BANK_VAULT_ZONE.x, Player.z - BANK_VAULT_ZONE.z);

  } else {
    logEvent(`⚙️ Wiercenie trwa: ${Math.floor(GameState.heistProgress)}%...`, 'text-cyan-300');
  }
}

function fireCurrentWeapon3D() {
  if (Player.reloading || Player.fireCooldown > 0) return;
  const w = Player.weapons[Player.currentWeaponIdx];
  if (!w) return;

  if (w.currentClip <= 0 && w.clip !== Infinity) {
    startReload();
    return;
  }

  Player.fireCooldown = w.fireRate;
  if (w.clip !== Infinity) w.currentClip--;

  const isSilenced = Player.suppressor && (w.type === 'pistol' || w.type === 'smg' || w.type === 'rifle');
  SFX.shoot(isSilenced ? 'silenced' : w.type);

  const dirX = Math.sin(Player.camYaw);
  const dirZ = Math.cos(Player.camYaw);
  const startX = Player.x + dirX * 1.4;
  const startY = Player.y + 1.45;
  const startZ = Player.z + dirZ * 1.4;

  if (w.type === 'melee') {
    spawnParticles3D(startX, startY, startZ, 6, 0xf8fafc, 8, 0.25);
    npcs3D.forEach(n => {
      if (n.hp > 0 && Math.hypot(n.x - startX, n.z - startZ) < w.range) {
        damageNpc3D(n, w.dmg, 'player', dirX * 14, dirZ * 14);
        addWantedStars(1, 'Atak bronią białą');
      }
    });
    return;
  }

  const pellets = w.pellets || 1;
  const spread = w.spread || 0.025;
  for (let i = 0; i < pellets; i++) {
    const yawSpread = Player.camYaw + (Math.random() - 0.5) * spread;
    const vx = Math.sin(yawSpread) * (w.type === 'rpg' ? 85 : w.type === 'flame' ? 45 : 165);
    const vz = Math.cos(yawSpread) * (w.type === 'rpg' ? 85 : w.type === 'flame' ? 45 : 165);

    const bMesh = new THREE.Mesh(
      new THREE.SphereGeometry(w.explosive ? 0.35 : w.type === 'flame' ? 0.45 : 0.16, 6, 6),
      new THREE.MeshBasicMaterial({ color: w.type === 'flame' ? 0xf97316 : 0xfde047 })
    );
    bMesh.position.set(startX, startY, startZ);
    scene.add(bMesh);

    bullets3D.push({
      x: startX, y: startY, z: startZ,
      vx, vy: 0, vz,
      dmg: w.dmg,
      range: w.range,
      traveled: 0,
      owner: 'player',
      explosive: !!w.explosive,
      mesh: bMesh
    });
  }

  if (!isSilenced) addWantedStars(1, `Strzały z ${w.name}`);
  updateHud();
}

  const v = Player.inVehicle;
  if (!v || v.fireCooldown > 0) return;
  v.fireCooldown = 1.1;
  SFX.shoot('tank');

  const dirZ = Math.cos(v.yaw);

  bMesh.position.set(sx, 1.8, sz);
  scene.add(bMesh);

  bullets3D.push({
    x: sx, y: 1.8, z: sz,
    vx: dirX * 110, vy: 0, vz: dirZ * 110,
    dmg: 350,
    range: 180,
    traveled: 0,
    owner: 'player',
    explosive: true,
    mesh: bMesh
  });

}

function createExplosion3D(x, y, z, radius = 18, maxDmg = 260) {
  SFX.explosion();
  spawnParticles3D(x, y + 1, z, 28, 0xf97316, 26, 0.65);
  spawnParticles3D(x, y + 1, z, 18, 0xfacc15, 18, 0.5);

  if (Math.hypot(Player.x - x, Player.z - z) < radius) {
    damagePlayer(maxDmg * 0.35);
  }
  npcs3D.forEach(n => {
    const d = Math.hypot(n.x - x, n.z - z);
    if (n.hp > 0 && d < radius) {
      const kx = ((n.x - x) / Math.max(0.1, d)) * 22;
      const kz = ((n.z - z) / Math.max(0.1, d)) * 22;
      damageNpc3D(n, maxDmg * (1 - d / radius), 'player', kx, kz);
    }
  });
  vehicles3D.forEach(v => {
    const d = Math.hypot(v.x - x, v.z - z);
    if (d < radius) v.hp -= maxDmg * 1.15 * (1 - d / radius);
  });
}

function spawnParticles3D(x, y, z, count, colorHex, speed = 12, life = 0.5) {
  const mat = new THREE.MeshBasicMaterial({ color: colorHex });
  const geo = new THREE.BoxGeometry(0.35, 0.35, 0.35);
  for (let i = 0; i < count; i++) {
    const m = new THREE.Mesh(geo, mat);
    m.position.set(x, y, z);
    scene.add(m);
    const a = Math.random() * Math.PI * 2;
    const s = Math.random() * speed;
    particles3D.push({
      mesh: m,
      vx: Math.cos(a) * s,
      vy: 3 + Math.random() * speed * 0.6,
      vz: Math.sin(a) * s,
      life
    });
  }
}

function startReload() {
  const w = Player.weapons[Player.currentWeaponIdx];
  if (!w || w.clip === Infinity || w.currentClip === w.clip || w.currentReserve <= 0 || Player.reloading) return;
  Player.reloading = true;
  Player.reloadTimer = 1.0;
  logEvent(`🔄 Przeładowywanie 3D: ${w.name}...`, 'text-slate-400');
}

function damageNpc3D(npc, dmg, attacker, knockX = 0, knockZ = 0) {
  npc.hp -= dmg;
  npc.vx += knockX;
  npc.vz += knockZ;
  npc.vy = Math.max(npc.vy, 6.2);
  npc.ragdollTimer = 1.25;
  npc.mesh.rotation.x = Math.PI * 0.45;
  if (npc.limbLeftLeg) npc.limbLeftLeg.rotation.x = 0.7;
  if (npc.limbRightLeg) npc.limbRightLeg.rotation.x = -0.6;
  spawnParticles3D(npc.x, 1.2, npc.z, 5, 0xdc2626, 7, 0.35);

  if (npc.hp <= 0) {
    npc.hp = 0;
    npc.state = 'dead';
    npc.mesh.rotation.x = Math.PI / 2;
    npc.posY = 0.25;
    if (attacker === 'player') {
      Player.kills++;
      Player.cash += npc.cash;
      SFX.cash();
      logEvent(`💀 Wyeliminowano: ${npc.name} (+$${npc.cash})`, 'text-emerald-400');
      if (npc.role === 'police' || npc.role === 'swat') addWantedStars(1, 'Zabójstwo policjanta');
      if (npc.role === 'fbi' || npc.role === 'army') addWantedStars(2, 'Zabójstwo agenta FBI/Wojska');
    }
  } else if (attacker === 'player') {
    npc.state = npc.weapon ? 'combat' : 'flee';
  }
}

function damagePlayer(dmg) {
  if (Player.armor > 0) {
    const absorbed = Math.min(Player.armor, dmg * 0.7);
    Player.armor -= absorbed;
    dmg -= absorbed;
  }
  Player.hp = Math.max(0, Player.hp - dmg);
  if (Player.hp <= 0) {
    Player.hp = 100;
    Player.armor = 0;
    Player.inVehicle = null;
    Player.mesh.visible = true;
    document.getElementById('vehicleHud').classList.add('hidden');
    const fee = Math.min(Player.cash, 200);
    Player.cash -= fee;
    GameState.wantedLevel = 0;
    Player.x = -38;
    Player.z = 52;
    Player.y = 0;
    logEvent(`🏥 WASTED! Odrodziłeś się przed Szpitalem Centralnym (-$${fee}).`, 'text-red-400 font-bold');
  }
}

    const dx = e.clientX - Mouse.lastX;
    const dy = e.clientY - Mouse.lastY;
    Mouse.lastX = e.clientX;
    Mouse.lastY = e.clientY;
    const sens = Settings.camSens * 0.0014;
    Player.camYaw -= dx * sens;
    Player.camPitch = Math.max(0.06, Math.min(1.15, Player.camPitch + dy * sens));
  }
});

function getNearbyNpc(maxDist = 7.5) {
  let best = null;
  let bestD = maxDist;
  npcs3D.forEach(n => {
    if (n.hp > 0) {
      const d = Math.hypot(n.x - Player.x, n.z - Player.z);
      if (d < bestD) { bestD = d; best = n; }
    }
  });
  return best;
}

function openNpcChatModal() {
  const npc = getNearbyNpc(8.5);
  if (!npc) {
    logEvent('💬 Podejdź bliżej dowolnego NPC (do 8.5m), aby porozmawiać [T].', 'text-amber-300');

function handleEnterInteractOrBuilding3D() {
  if (Player.inVehicle) {
    const v = Player.inVehicle;
    Player.x = v.x + Math.cos(v.yaw) * (v.w + 2.4);
    Player.z = v.z - Math.sin(v.yaw) * (v.w + 2.4);
    Player.y = Math.max(0, v.y);
    Player.vy = 0;
    v.driver = null;
    Player.inVehicle = null;
    Player.mesh.visible = true;
    document.getElementById('vehicleHud').classList.add('hidden');
    logEvent(`🚪 Wysiadłeś z: ${v.name}`, 'text-slate-300');
    return;
  }

  let closestVeh = null;
  let bestDist = 8.5;
  vehicles3D.forEach(v => {
    const d = Math.hypot(v.x - Player.x, v.z - Player.z);
    if (d < bestDist) { bestDist = d; closestVeh = v; }
  });

  if (closestVeh) {
    const wasAiDriven = !!closestVeh.aiDriver;
    closestVeh.aiDriver = null;
    Player.inVehicle = closestVeh;
    closestVeh.driver = 'player';
    Player.mesh.visible = false;
    document.getElementById('vehicleHud').classList.remove('hidden');
    SFX.init();
    if (wasAiDriven) {
      addWantedStars(1, `Kradzież pojazdu: ${closestVeh.name}`);
      logEvent(`🔓 Ukradłeś i wsiadłeś do 3D: ${closestVeh.name} [${closestVeh.plate}]!`, 'text-amber-300 font-bold');
    } else {
      logEvent(`🔑 Wsiadłeś do 3D: ${closestVeh.name} [${closestVeh.plate}]`, 'text-cyan-300');
    }
    return;
  }

  const nearB = getNearbyBuilding3D();
  if (nearB) {
    enterBuilding3D(nearB);
  }
}

function getCurrentCityName() {
  if (Math.hypot(Player.x - PARK_ZONE.x, Player.z - PARK_ZONE.z) < PARK_ZONE.r) return PARK_ZONE.name;
  for (const c of CITIES) {
    if (Math.hypot(Player.x - c.x, Player.z - c.z) < c.r) return c.name;
  }
  if (Player.x > 310) return 'Ocean Spokojny SUS 3D';
  return 'Autostrada Stanowa SUS 3D';
}

function collidesWithSolid3D(x, z, r = 0.8) {
  for (const c of solidColliders) {
    if (Math.abs(x - c.x) < c.hw + r && Math.abs(z - c.z) < c.hd + r) return true;
  }
  return false;
}

function isPointInWater3D(x, z) {
  for (const w of WATER_ZONES) {
    if (Math.abs(x - w.x) < w.w / 2 && Math.abs(z - w.z) < w.d / 2) return true;
  }
  return false;
}

function updateSimulation3D(dt) {
  if (!GameState.running || GameState.paused) return;

  GameState.timeMinutes = (GameState.timeMinutes + dt * 2) % (24 * 60);

  if (GameState.heistActive) {
    const distVault = Math.hypot(Player.x - BANK_VAULT_ZONE.x, Player.z - BANK_VAULT_ZONE.z);
    const distBank = Math.hypot(Player.x - 38, Player.z - 36);
    if (distVault > 34 && distBank > 34) {
      GameState.heistActive = false;
      GameState.heistProgress = 0;
      logEvent('⚠️ Oddaliłeś się zbyt daleko od banku — wiercenie przerwane.', 'text-amber-400');
    } else {
      GameState.heistProgress = Math.min(100, GameState.heistProgress + 16 * dt);
      if (GameState.heistProgress >= 100) {
        GameState.heistActive = false;
        GameState.heistLootReady = Math.floor(2800 + Math.random() * 2400);
        logEvent(`🔓 Skarbiec otwarty! Łup $${GameState.heistLootReady} czeka — wciśnij [H], aby go zabrać!`, 'text-emerald-400 font-bold');
      }
    }
  }

  if (Player.reloading) {
    Player.reloadTimer -= dt;
    if (Player.reloadTimer <= 0) {
      Player.reloading = false;
      const w = Player.weapons[Player.currentWeaponIdx];
      if (w && w.clip !== Infinity) {
        const need = w.clip - w.currentClip;
        const take = Math.min(need, w.currentReserve);
        w.currentClip += take;
        w.currentReserve -= take;
      }
    }
  }
  if (Player.fireCooldown > 0) Player.fireCooldown -= dt;

  if (Mouse.down && !Player.inVehicle) {
    const w = Player.weapons[Player.currentWeaponIdx];
    if (w && w.auto) fireCurrentWeapon3D();
  }

  if (Keys['ArrowLeft']) Player.camYaw += 2.2 * dt;
  if (Keys['ArrowRight']) Player.camYaw -= 2.2 * dt;

  if (!Player.inVehicle) {
    updatePlayerOnFoot3D(dt);
  } else {
    updatePlayerInVehicle3D(dt);
  }

  if (Player.hasDog && dogMesh) {
    const tx = Player.x - Math.sin(Player.camYaw) * 2.5;
    const tz = Player.z - Math.cos(Player.camYaw) * 2.5;
    dogMesh.position.x += (tx - dogMesh.position.x) * 4 * dt;
    dogMesh.position.z += (tz - dogMesh.position.z) * 4 * dt;

    const enemy = npcs3D.find(n => n.hp > 0 && n.state === 'combat' && Math.hypot(n.x - Player.x, n.z - Player.z) < 22);
    if (enemy) {
      dogMesh.position.x += (enemy.x - dogMesh.position.x) * 5 * dt;
      dogMesh.position.z += (enemy.z - dogMesh.position.z) * 5 * dt;
      if (Math.hypot(enemy.x - dogMesh.position.x, enemy.z - dogMesh.position.z) < 2.2) {
        damageNpc3D(enemy, 32 * dt, 'player');
      }
    }
  }

  if (GameState.witnessCallingNpc) {
    const wn = GameState.witnessCallingNpc;
    if (wn.hp <= 0 || Math.hypot(wn.x - Player.x, wn.z - Player.z) > 75) {
      logEvent('📵 Połączenie z 911 przerwane!', 'text-emerald-400');
      GameState.witnessCallingNpc = null;
    } else {
      GameState.witnessCallTimer -= dt;
      if (GameState.witnessCallTimer <= 0) {
        GameState.witnessCallingNpc = null;
        addWantedStars(1, 'Zgłoszenie świadka 911');
      }
    }
  }

  if (GameState.wantedLevel > 0) {
    const lawNear = npcs3D.some(n => n.hp > 0 && ['police', 'swat', 'fbi', 'army'].includes(n.role) && Math.hypot(n.x - Player.x, n.z - Player.z) < 75);
    if (!lawNear) {
      GameState.wantedDecayTimer -= dt;
      if (GameState.wantedDecayTimer <= 0) {
        GameState.wantedLevel = Math.max(0, GameState.wantedLevel - 1);
        GameState.wantedDecayTimer = 14;
        if (GameState.wantedLevel === 0) logEvent('✅ Zgubiłeś pościg! Poziom poszukiwania: 0★.', 'text-emerald-400 font-bold');
      }
    } else {
      GameState.wantedDecayTimer = 16;
      if (Math.random() < dt * 0.2 && npcs3D.filter(n => n.hp > 0 && ['police', 'swat', 'fbi', 'army'].includes(n.role)).length < GameState.wantedLevel * 3) {
        spawnPursuitUnits3D();
      }
    }
  }

  updateVehicles3D(dt);
  updateNpcs3D(dt);
  updateBullets3D(dt);
  updateParticles3D(dt);

  if (fountainStreamMesh) {
    const pulse = 1 + Math.sin(performance.now() * 0.008) * 0.22;
    fountainStreamMesh.scale.set(pulse, pulse, pulse);
  }
  clouds3D.forEach(cl => {
    cl.mesh.position.x += cl.speed * dt;
    if (cl.mesh.position.x > WORLD_HALF + 60) cl.mesh.position.x = -WORLD_HALF - 60;
  });

  trainX += 32 * dt;
  if (trainX > WORLD_HALF + 80) trainX = -WORLD_HALF - 40;
  if (trainMesh) trainMesh.position.x = trainX;

  if (aiPlaneMesh) {
    if (aiPlaneState.phase === 'taxi') {
      aiPlaneState.x -= 14 * dt;
      if (aiPlaneState.x < 210) aiPlaneState.phase = 'takeoff';
    } else {
      aiPlaneState.x -= 48 * dt;
      aiPlaneState.y = Math.min(55, aiPlaneState.y + 12 * dt);
      if (aiPlaneState.x < -420) {
        aiPlaneState.x = 320;
        aiPlaneState.y = 1.5;
        aiPlaneState.phase = 'taxi';
      }
    }
    aiPlaneMesh.position.set(aiPlaneState.x, aiPlaneState.y, aiPlaneState.z);
  }

  if (aiHeliMesh) {
    aiHeliState.angle += 0.28 * dt;
    const hx = Math.cos(aiHeliState.angle) * aiHeliState.radius;
    const hz = Math.sin(aiHeliState.angle) * aiHeliState.radius;
    aiHeliMesh.position.set(hx, aiHeliState.y, hz);
    aiHeliMesh.rotation.y = -aiHeliState.angle;
    if (aiHeliMesh.userData.rotor) aiHeliMesh.userData.rotor.rotation.y += 28 * dt;
  }

  updateEnvironment3D(dt);
  updateCamera3D();
  SFX.updateRadio(dt, !!Player.inVehicle);
  updateHud();
}

function updatePlayerOnFoot3D(dt) {
  let forward = 0, strafe = 0;
  if (Keys['KeyW'] || Keys['ArrowUp']) forward += 1;
  if (Keys['KeyS']) forward -= 1;
  if (Keys['KeyA']) strafe += 1;
  if (Keys['KeyD']) strafe -= 1;

  const sprinting = (Keys['ShiftLeft'] || Keys['ShiftRight']) && Player.stamina > 5;
  const maxSpd = sprinting ? 19.5 : 12.0;

  if (sprinting && (forward !== 0 || strafe !== 0)) Player.stamina = Math.max(0, Player.stamina - 20 * dt);
  else Player.stamina = Math.min(100, Player.stamina + 15 * dt);

  if (Keys['Space'] && Player.y <= 0.05 && Player.rollTimer <= 0) {
    Player.vy = 9.8;
  }
  Player.vy -= 24 * dt;
  const prevY = Player.y;
  Player.y = Math.max(0, Player.y + Player.vy * dt);
  if (Player.y === 0) {
    if (prevY > 8.5 && Player.vy < -16) {
      const fallDmg = Math.min(45, Math.round((Math.abs(Player.vy) - 16) * 2.1));
      damagePlayer(fallDmg);
      Player.rollTimer = 0.35;
      logEvent(`💥 Upadek z wysokości (-${fallDmg} HP)!`, 'text-amber-400');
    }
    Player.vy = 0;
  }

  const fx = Math.sin(Player.camYaw);
  const fz = Math.cos(Player.camYaw);
  const rx = Math.cos(Player.camYaw);
  const rz = -Math.sin(Player.camYaw);

  let mx = fx * forward + rx * strafe;
  let mz = fz * forward + rz * strafe;
  const len = Math.hypot(mx, mz);
  if (len > 0) { mx /= len; mz /= len; }

  if (Player.rollTimer > 0) {
    Player.rollTimer -= dt;
    Player.mesh.rotation.x += 14 * dt;
  } else {
    Player.mesh.rotation.x = 0;
    const targetVx = mx * maxSpd;
    const targetVz = mz * maxSpd;
    const smooth = 1 - Math.exp(-14 * dt);
    Player.vx += (targetVx - Player.vx) * smooth;
    Player.vz += (targetVz - Player.vz) * smooth;
  }

  const nx = Math.max(-WORLD_HALF + 5, Math.min(WORLD_HALF - 5, Player.x + Player.vx * dt));
  const nz = Math.max(-WORLD_HALF + 5, Math.min(WORLD_HALF - 5, Player.z + Player.vz * dt));

  if (!collidesWithSolid3D(nx, Player.z, 0.7)) Player.x = nx;
  else Player.vx = 0;
  if (!collidesWithSolid3D(Player.x, nz, 0.7)) Player.z = nz;
  else Player.vz = 0;

  if (len > 0.05) {
    const targetYaw = Math.atan2(mx, mz);
    let diff = targetYaw - Player.yaw;
    while (diff > Math.PI) diff -= Math.PI * 2;
    while (diff < -Math.PI) diff += Math.PI * 2;
    Player.yaw += diff * Math.min(1, 14 * dt);
  }

  if (Player.mesh) {
    Player.mesh.position.set(Player.x, Player.y, Player.z);
    Player.mesh.rotation.y = Player.yaw;
  }
}

function updatePlayerInVehicle3D(dt) {
  const v = Player.inVehicle;
  if (v.fireCooldown > 0) v.fireCooldown -= dt;

  const hasFuel = !Settings.fuelEnabled || v.fuel > 0;
  let throttle = 0;
  if (Keys['KeyW'] || Keys['ArrowUp']) throttle += 1;
  if (Keys['KeyS'] || Keys['ArrowDown']) throttle -= 0.7;

  const nitro = (Keys['ShiftLeft'] || Keys['ShiftRight']) && hasFuel && throttle > 0;

  if (hasFuel && throttle !== 0) {
    v.speed += throttle * v.accel * (nitro ? 1.45 : 1.0) * dt;
    if (Settings.fuelEnabled) v.fuel = Math.max(0, v.fuel - (nitro ? 1.5 : 0.45) * dt);
  } else {
    v.speed *= Math.pow(0.32, dt);
  }

  if (Keys['Space'] && !v.flying) v.speed *= Math.pow(0.12, dt);

  const maxSpd = nitro ? v.maxSpeed * 1.25 : v.maxSpeed;
  v.speed = Math.max(-maxSpd * 0.45, Math.min(maxSpd, v.speed));

  if (Math.abs(v.speed) > 1.0 || v.flying) {
    const dir = v.speed >= 0 ? 1 : -1;
    if (Keys['KeyA']) v.yaw += v.turn * dir * dt;
    if (Keys['KeyD']) v.yaw -= v.turn * dir * dt;
  }

  if (v.heli) {
    if (Keys['KeyR']) v.y = Math.min(95, v.y + 22 * dt);
    if (Keys['KeyQ']) v.y = Math.max(0, v.y - 22 * dt);
  } else if (v.airplane) {
    if (v.speed > 18 && Keys['KeyR']) v.y = Math.min(115, v.y + (v.speed * 0.35) * dt);
    else if (Keys['KeyQ'] || v.speed < 14) v.y = Math.max(0, v.y - 16 * dt);
    v.pitch = v.y > 0.5 ? (Keys['KeyR'] ? -0.16 : Keys['KeyQ'] ? 0.14 : -0.02) : 0;
  }

  const nx = Math.max(-WORLD_HALF + 8, Math.min(WORLD_HALF - 8, v.x + Math.sin(v.yaw) * v.speed * dt));
  const nz = Math.max(-WORLD_HALF + 8, Math.min(WORLD_HALF - 8, v.z + Math.cos(v.yaw) * v.speed * dt));

  const inWater = isPointInWater3D(nx, nz);
  if (inWater && !v.waterOnly && !v.flying) {
    v.hp -= 55 * dt;
    v.speed *= 0.85;
  }
  if (!inWater && v.waterOnly && v.y <= 0.1) {
    v.speed *= 0.65;
  }

  if (v.flying || !collidesWithSolid3D(nx, nz, Math.min(v.w, v.l) * 0.45)) {
    v.x = nx;
    v.z = nz;
  } else {
    v.hp -= Math.abs(v.speed) * 0.6;
    v.speed *= -0.35;
    SFX.playTone(110, 'sawtooth', 0.12, 0.12);
  }

  if (nitro) {
    spawnParticles3D(v.x - Math.sin(v.yaw) * (v.l * 0.55), v.y + 0.6, v.z - Math.cos(v.yaw) * (v.l * 0.55), 2, 0x38bdf8, 6, 0.2);
  }

  v.mesh.position.set(v.x, v.y, v.z);
  v.mesh.rotation.y = v.yaw;
  v.mesh.rotation.x = v.pitch || 0;

  Player.x = v.x;
  Player.y = v.y;
  Player.z = v.z;
  Player.camYaw = v.yaw;

  if (v.sirenOn) SFX.sirenTick(performance.now() / 1000);
}

function updateVehicles3D(dt) {
  for (let i = vehicles3D.length - 1; i >= 0; i--) {
    const v = vehicles3D[i];
    if (v.rotorMesh) v.rotorMesh.rotation.y += (v.driver ? 26 : 4) * dt;
    if (v.sirenMesh && v.sirenOn) {
      const flash = Math.floor(performance.now() / 180) % 2 === 0;
      v.sirenMesh.material.color.setHex(flash ? 0xef4444 : 0x3b82f6);
    }

    // Ruch AI pojazdów drogowych oraz patrolującego czołgu AI
    if (v.aiDriver && !v.driver && v.hp > 0) {
      const spd = v.aiDriver === 'tank_patrol' ? 11.5 : 18.0;
      if (v.patrolAxis === 'x') {
        v.x += v.patrolDir * spd * dt;
        v.yaw = v.patrolDir > 0 ? Math.PI / 2 : -Math.PI / 2;
        if (Math.abs(v.x) > 190) v.patrolDir *= -1;
      } else {
        v.z += v.patrolDir * spd * dt;
        v.yaw = v.patrolDir > 0 ? 0 : Math.PI;
        if (Math.abs(v.z) > 190) v.patrolDir *= -1;
      }
      v.mesh.position.set(v.x, v.y, v.z);
      v.mesh.rotation.y = v.yaw;
    }

    if (v.hp < v.maxHp * 0.45 && v.hp > 0 && Math.random() < 0.25) {
      spawnParticles3D(v.x, v.y + 1.4, v.z, 1, v.hp < v.maxHp * 0.2 ? 0xf97316 : 0x475569, 4, 0.5);
    }

    if (v.hp <= 0) {
      createExplosion3D(v.x, v.y, v.z, 18, 250);
      if (Player.inVehicle === v) {
        Player.inVehicle = null;
        Player.mesh.visible = true;
        document.getElementById('vehicleHud').classList.add('hidden');
      }
      scene.remove(v.mesh);
      vehicles3D.splice(i, 1);
      continue;
    }

    if (Math.abs(v.speed) > 12 && v.y < 1.2) {
      npcs3D.forEach(n => {
        if (n.hp > 0 && Math.hypot(n.x - v.x, n.z - v.z) < v.w * 0.9) {
          damageNpc3D(n, Math.abs(v.speed) * 2.5, v.driver === 'player' ? 'player' : 'ai', Math.sin(v.yaw) * 18, Math.cos(v.yaw) * 18);
          if (v.driver === 'player') addWantedStars(1, 'Potrącenie pieszego');
        }
      });
      props3D.forEach(p => {
        if (!p.broken && Math.hypot(p.x - v.x, p.z - v.z) < v.w * 0.9) {
          p.broken = true;
          p.mesh.rotation.z = Math.PI / 2;
          p.mesh.position.y = 0.2;
          if (p.isHydrant) spawnParticles3D(p.x, 1, p.z, 22, 0x38bdf8, 16, 0.8);
        }
      });
    }
  }
}

function updateNpcs3D(dt) {
  npcs3D.forEach(n => {
    if (n.hp <= 0) return;
    if (n.fireCooldown > 0) n.fireCooldown -= dt;

    if (n.ragdollTimer > 0 || n.posY > 0 || Math.hypot(n.vx, n.vz) > 0.5) {
      n.ragdollTimer = Math.max(0, n.ragdollTimer - dt);
      n.vy -= 20 * dt;
      n.posY = Math.max(0, n.posY + n.vy * dt);
      if (n.posY === 0 && n.vy < 0) n.vy = 0;
      n.x = Math.max(-WORLD_HALF + 4, Math.min(WORLD_HALF - 4, n.x + n.vx * dt));
      n.z = Math.max(-WORLD_HALF + 4, Math.min(WORLD_HALF - 4, n.z + n.vz * dt));
      n.vx *= Math.pow(0.08, dt);
      n.vz *= Math.pow(0.08, dt);
      if (n.ragdollTimer <= 0 && n.posY === 0) {
        n.mesh.rotation.x = 0;
        if (n.limbLeftLeg) n.limbLeftLeg.rotation.x = 0;
        if (n.limbRightLeg) n.limbRightLeg.rotation.x = 0;
      }
    }

    const dist = Math.hypot(Player.x - n.x, Player.z - n.z);
    const isLaw = ['police', 'swat', 'fbi', 'army'].includes(n.role);

    if (isLaw && GameState.wantedLevel > 0 && dist < 95) n.state = 'combat';
    else if (isLaw && GameState.wantedLevel === 0) n.state = 'wander';

    if (n.state !== 'call911' && n.ragdollTimer <= 0) {
      if (n.state === 'flee') {
        n.yaw = Math.atan2(n.x - Player.x, n.z - Player.z);
        n.x += Math.sin(n.yaw) * 11 * dt;
        n.z += Math.cos(n.yaw) * 11 * dt;
      } else if (n.state === 'combat') {
        n.yaw = Math.atan2(Player.x - n.x, Player.z - n.z);
        if (dist > 22) {
          const spd = n.role === 'fbi' ? 12.5 : 10;
          n.x += Math.sin(n.yaw) * spd * dt;
          n.z += Math.cos(n.yaw) * spd * dt;
        }
        if (n.weapon && dist < n.weapon.range && n.fireCooldown <= 0) {
          n.fireCooldown = n.role === 'fbi' || n.role === 'swat' ? 0.5 : 0.9;
          const bMesh = new THREE.Mesh(new THREE.SphereGeometry(0.16, 6, 6), new THREE.MeshBasicMaterial({ color: 0xef4444 }));
          bMesh.position.set(n.x, 1.4, n.z);
          scene.add(bMesh);
          bullets3D.push({
            x: n.x, y: 1.4, z: n.z,
            vx: Math.sin(n.yaw) * 130,
            vy: 0,
            vz: Math.cos(n.yaw) * 130,
            dmg: n.role === 'fbi' ? 13 : 8,
            range: n.weapon.range,
            traveled: 0,
            owner: 'law',
            mesh: bMesh
          });
        }
      } else {
        n.timer -= dt;
        if (n.timer <= 0) {
          n.timer = 2 + Math.random() * 4;
          n.yaw += (Math.random() - 0.5) * 2.5;
        }
        const spd = 3.8;
        const nx = n.x + Math.sin(n.yaw) * spd * dt;
        const nz = n.z + Math.cos(n.yaw) * spd * dt;
        if (!collidesWithSolid3D(nx, nz, 0.7) && !isPointInWater3D(nx, nz)) {
          n.x = nx; n.z = nz;
          const swing = Math.sin(performance.now() * 0.012) * 0.45;
          if (n.limbLeftLeg) n.limbLeftLeg.rotation.x = swing;
          if (n.limbRightLeg) n.limbRightLeg.rotation.x = -swing;
        } else {
          n.yaw += Math.PI * 0.7;
        }
      }
    }

    n.mesh.position.set(n.x, n.posY, n.z);
    n.mesh.rotation.y = n.yaw;
  });
}

function updateBullets3D(dt) {
  for (let i = bullets3D.length - 1; i >= 0; i--) {
    const b = bullets3D[i];
    const sx = b.vx * dt;
    const sz = b.vz * dt;
    b.x += sx;
    b.z += sz;
    b.traveled += Math.hypot(sx, sz);
    b.mesh.position.set(b.x, b.y, b.z);

    let hit = collidesWithSolid3D(b.x, b.z, 0.2) || b.traveled >= b.range;

    if (!hit && b.owner !== 'player' && Math.hypot(b.x - Player.x, b.z - Player.z) < 1.1) {
      hit = true;
      if (Player.inVehicle) Player.inVehicle.hp -= b.dmg;
      else damagePlayer(b.dmg);
    }

    if (!hit) {
      for (const n of npcs3D) {
        if (n.hp > 0 && Math.hypot(b.x - n.x, b.z - n.z) < 1.1) {
          hit = true;
          damageNpc3D(n, b.dmg, b.owner, b.vx * 0.08, b.vz * 0.08);
          break;
        }
      }
    }

    if (!hit) {
      for (const v of vehicles3D) {
        if (v !== Player.inVehicle && Math.hypot(b.x - v.x, b.z - v.z) < Math.max(v.w, v.l) * 0.5) {
          hit = true;
          v.hp -= b.dmg;
          spawnParticles3D(b.x, b.y, b.z, 4, 0xfacc15, 8, 0.25);
          break;
        }
      }
    }

    if (hit) {
      if (b.explosive) createExplosion3D(b.x, b.y, b.z, 18, b.dmg);
      scene.remove(b.mesh);
      bullets3D.splice(i, 1);
    }
  }
}

function updateParticles3D(dt) {
  for (let i = particles3D.length - 1; i >= 0; i--) {
    const p = particles3D[i];
    p.mesh.position.x += p.vx * dt;
    p.mesh.position.y += p.vy * dt;
    p.mesh.position.z += p.vz * dt;
    p.vy -= 18 * dt;
    p.life -= dt;
    if (p.life <= 0) {
      scene.remove(p.mesh);
      particles3D.splice(i, 1);
    }
  }
}

function updateEnvironment3D(dt) {
  applyGraphicsPreset();
  syncCloudDensity();

  const seasonGround = { spring: 0x1e3a29, summer: 0x254228, autumn: 0x452b1b, winter: 0xe2e8f0 };
  if (groundMesh) groundMesh.material.color.setHex(seasonGround[Settings.season] || 0x1e3a29);

  const isNightOrSunset = Settings.timeOfDay === 'night' || Settings.timeOfDay === 'sunset';
  nightWindows3D.forEach(w => { w.visible = isNightOrSunset; });
  streetLamps3D.forEach(m => { m.color.setHex(isNightOrSunset ? 0xfef08a : 0x94a3b8); });

  if (Settings.timeOfDay === 'night') {
    scene.background.setHex(0x040712);
    scene.fog.color.setHex(0x040712);
    dirLight.intensity = 0.28;
    ambientLight.intensity = 0.35;
  } else if (Settings.timeOfDay === 'sunset') {
    scene.background.setHex(0x451a2b);
    scene.fog.color.setHex(0x451a2b);
    dirLight.intensity = 0.75;
    ambientLight.intensity = 0.55;
  } else {
    scene.background.setHex(0x38bdf8);
    scene.fog.color.setHex(0x38bdf8);
    dirLight.intensity = 1.2;
    ambientLight.intensity = 0.78;
  }

  scene.fog.density = Settings.weather === 'fog' ? 0.011 : Settings.weather === 'storm' ? 0.006 : 0.0018;
  if (weatherParticles) {
    const activeWeather = ['rain', 'storm', 'snow'].includes(Settings.weather) || Settings.season === 'winter';
    weatherParticles.visible = activeWeather;
    if (activeWeather) {
      const isSnow = Settings.weather === 'snow' || Settings.season === 'winter';
      weatherParticles.material.color.setHex(isSnow ? 0xffffff : 0x93c5fd);
      weatherParticles.position.set(Player.x, 0, Player.z);
      const pos = weatherParticles.geometry.attributes.position.array;
      for (let i = 1; i < pos.length; i += 3) {
        pos[i] -= (isSnow ? 14 : 45) * dt;
        if (pos[i] < 0) pos[i] = 75;
      }
      weatherParticles.geometry.attributes.position.needsUpdate = true;
    }
  }
}

function updateCamera3D() {
  if (Player.fppMode) {
    const eyeY = Player.inVehicle ? Player.y + 1.55 : Player.y + 2.05;
    camera.position.set(Player.x, eyeY, Player.z);
    const lookX = Player.x + Math.sin(Player.camYaw) * 30;
    const lookZ = Player.z + Math.cos(Player.camYaw) * 30;
    camera.lookAt(lookX, eyeY - (Player.camPitch - 0.34) * 14, lookZ);
  } else {
    const isAir = Player.inVehicle && Player.inVehicle.flying;
    const dist = isAir ? 18 : Player.inVehicle ? 14.5 : 8.5;
    const height = isAir ? 7.5 : Player.inVehicle ? 5.2 : 3.6;
    const targetCamX = Player.x - Math.sin(Player.camYaw) * dist;
    const targetCamZ = Player.z - Math.cos(Player.camYaw) * dist;
    const targetCamY = Player.y + height + Player.camPitch * 6;

    camera.position.lerp(new THREE.Vector3(targetCamX, targetCamY, targetCamZ), 0.16);
    camera.lookAt(Player.x, Player.y + 1.5, Player.z);
  }
}

function renderMinimap3D() {
  const w = miniCanvas.width;
  const h = miniCanvas.height;
  const scale = 0.65;

  miniCtx.fillStyle = '#070b14';
  miniCtx.fillRect(0, 0, w, h);

  miniCtx.save();
  miniCtx.translate(w / 2 - Player.x * scale, h / 2 - Player.z * scale);

  miniCtx.fillStyle = '#0284c7';
  WATER_ZONES.forEach(wz => miniCtx.fillRect((wz.x - wz.w / 2) * scale, (wz.z - wz.d / 2) * scale, wz.w * scale, wz.d * scale));

  miniCtx.fillStyle = '#16a34a';
  miniCtx.beginPath();
  miniCtx.arc(PARK_ZONE.x * scale, PARK_ZONE.z * scale, PARK_ZONE.r * scale, 0, Math.PI * 2);
  miniCtx.fill();

  BUILDINGS.forEach(b => {
    miniCtx.fillStyle = '#' + b.color.toString(16).padStart(6, '0');
    miniCtx.fillRect((b.x - b.w / 2) * scale, (b.z - b.d / 2) * scale, Math.max(5, b.w * scale), Math.max(5, b.d * scale));
  });

  npcs3D.forEach(n => {
    if (n.hp <= 0) return;
    if (['police', 'swat', 'fbi', 'army'].includes(n.role)) {
      miniCtx.fillStyle = Math.floor(performance.now() / 180) % 2 === 0 ? '#ef4444' : '#3b82f6';
      miniCtx.beginPath();
      miniCtx.arc(n.x * scale, n.z * scale, 3, 0, Math.PI * 2);
      miniCtx.fill();
    }
  });

  if (GameState.gpsTarget) {
    miniCtx.fillStyle = '#ec4899';
    miniCtx.beginPath();
    miniCtx.arc(GameState.gpsTarget.x * scale, GameState.gpsTarget.z * scale, 5, 0, Math.PI * 2);
    miniCtx.fill();
  }

  miniCtx.restore();

  miniCtx.fillStyle = '#22c55e';
  miniCtx.beginPath();
  miniCtx.arc(w / 2, h / 2, 4.5, 0, Math.PI * 2);
  miniCtx.fill();
}

function renderFullMap() {
  const w = fullMapCanvas.width;
  const h = fullMapCanvas.height;
  const toMapX = x => ((x + WORLD_HALF) / (WORLD_HALF * 2)) * w;
  const toMapY = z => ((z + WORLD_HALF) / (WORLD_HALF * 2)) * h;

  fullMapCtx.fillStyle = Settings.season === 'winter' ? '#cbd5e1' : '#163020';
  fullMapCtx.fillRect(0, 0, w, h);

  fullMapCtx.fillStyle = '#0284c7';
  WATER_ZONES.forEach(wz => {
    fullMapCtx.fillRect(toMapX(wz.x - wz.w / 2), toMapY(wz.z - wz.d / 2), (wz.w / (WORLD_HALF * 2)) * w, (wz.d / (WORLD_HALF * 2)) * h);
  });

  fullMapCtx.fillStyle = '#16a34a';
  fullMapCtx.beginPath();
  fullMapCtx.arc(toMapX(PARK_ZONE.x), toMapY(PARK_ZONE.z), (PARK_ZONE.r / (WORLD_HALF * 2)) * w, 0, Math.PI * 2);
  fullMapCtx.fill();

  fullMapCtx.fillStyle = 'rgba(255,255,255,0.9)';
  fullMapCtx.font = 'bold 16px Bebas Neue, sans-serif';
  CITIES.forEach(c => fullMapCtx.fillText(c.name, toMapX(c.x) - 45, toMapY(c.z) - 25));

  BUILDINGS.forEach(b => {
    fullMapCtx.fillStyle = '#' + b.color.toString(16).padStart(6, '0');
    fullMapCtx.fillRect(toMapX(b.x) - 6, toMapY(b.z) - 6, 12, 12);
    fullMapCtx.fillStyle = '#ffffff';
    fullMapCtx.font = '11px Inter';
    fullMapCtx.fillText(`${b.icon} ${b.name}`, toMapX(b.x) + 8, toMapY(b.z) + 4);
  });

  fullMapCtx.fillStyle = '#22c55e';
  fullMapCtx.beginPath();
  fullMapCtx.arc(toMapX(Player.x), toMapY(Player.z), 7, 0, Math.PI * 2);
  fullMapCtx.fill();

  if (GameState.gpsTarget) {

    fullMapCtx.beginPath();
    fullMapCtx.arc(toMapX(GameState.gpsTarget.x), toMapY(GameState.gpsTarget.z), 10, 0, Math.PI * 2);
    fullMapCtx.stroke();
  }
}

function getFeatureRange(filter) {
  if (filter === 'ORIG') return [0, 100];
  if (filter === 'PREV350') return [100, 450];
  if (filter === 'NEW500') return [450, 950];
  if (filter === 'MEGA117900') return [950, 118850];
  if (filter === 'QOL50000') return [118850, 168850];
  if (filter === 'GFX55000') return [168850, 223850];
  if (filter === 'NEW1000000V7') return [223850, Encyclopedia2323850.totalCount];
  return [0, Encyclopedia2323850.totalCount];
}

function renderFeaturesList() {
  const query = document.getElementById('featureSearchInput').value.trim().toLowerCase();
  const filter = document.getElementById('featureCategoryFilter').value;
  const container = document.getElementById('featuresListContainer');
  container.innerHTML = '';

  const [startIdx, endIdx] = getFeatureRange(filter);
  const pageSize = 200;
  const totalInRange = endIdx - startIdx;
  const maxPage = Math.max(0, Math.floor((totalInRange - 1) / pageSize));
  if (GameState.featuresPage > maxPage) GameState.featuresPage = maxPage;

  const matches = [];
  if (!query) {
    const pStart = startIdx + GameState.featuresPage * pageSize;
    const pEnd = Math.min(endIdx, pStart + pageSize);
    for (let i = pStart; i < pEnd; i++) {
      matches.push(Encyclopedia2323850.getFeatureAt(i));
    }
    document.getElementById('featuresCountBadge').textContent = totalInRange.toLocaleString();
    document.getElementById('featuresPageLabel').textContent = `Strona ${GameState.featuresPage + 1} / ${maxPage + 1}`;
  } else {
    const found = [];
    for (let i = startIdx; i < endIdx; i++) {
      const item = Encyclopedia2323850.getFeatureAt(i);
      if (item.text.toLowerCase().includes(query) || item.cat.toLowerCase().includes(query)) {
        found.push(item);
        if (found.length >= 1200) break;
      }
    }
    const qMaxPage = Math.max(0, Math.floor((found.length - 1) / pageSize));
    if (GameState.featuresPage > qMaxPage) GameState.featuresPage = qMaxPage;
    const qStart = GameState.featuresPage * pageSize;
    matches.push(...found.slice(qStart, qStart + pageSize));
    document.getElementById('featuresCountBadge').textContent = `${found.length}${found.length >= 1200 ? '+' : ''}`;
    document.getElementById('featuresPageLabel').textContent = `Strona ${GameState.featuresPage + 1} / ${qMaxPage + 1}`;
  }

  matches.forEach(f => {
    const div = document.createElement('div');
    const borderCol = f.group === 'NEW1000000V7' ? 'border-amber-400/40' : f.group === 'QOL50000' ? 'border-emerald-500/30' : f.group === 'GFX55000' ? 'border-cyan-500/30' : 'border-pink-500/30';
    div.className = `p-2.5 rounded-xl bg-slate-900/95 border ${borderCol} flex justify-between items-start gap-2`;
    div.innerHTML = `
      <div>
        <span class="text-[10px] font-bold uppercase text-pink-400">${f.cat}</span>
        <div class="text-slate-200 mt-0.5">${f.text}</div>
      </div>
      <span class="px-2 py-0.5 rounded bg-emerald-500/20 text-emerald-300 text-[10px] font-bold shrink-0">3D ON</span>
    `;
    container.appendChild(div);
  });
}

function logEvent(msg, color = 'text-slate-200') {
  const box = document.getElementById('eventLog');

let lastTime = 0;
function animate3D(time) {
  const dt = Math.min(0.05, (time - lastTime) / 1000 || 0.016);
  lastTime = time;

  if (GameState.running) {
    updateSimulation3D(dt);
    renderMinimap3D();
    renderer.render(scene, camera);
  }
  requestAnimationFrame(animate3D);
}

window.onload = function () {
  requestAnimationFrame(animate3D);
};
